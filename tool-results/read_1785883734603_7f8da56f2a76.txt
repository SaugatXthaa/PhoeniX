    99→function hasPackKeyword(filenameLower) {
   100→    if (!filenameLower) return false;
   101→    const isRepack = /\brepack\b/.test(filenameLower);
   102→
   103→    return PACK_KEYWORDS.some(keyword => {
   104→        if (keyword === 'pack') {
   105→            return /\bpack\b/.test(filenameLower) && !isRepack;
   106→        }
   107→        return filenameLower.includes(keyword);
   108→    });
   109→}

   111→/**
   112→ * Check if a file is a pack based on filename and size
   113→ * @param {string} filename - The extracted filename
   114→ * @param {string} size - The file size (e.g., "5.9 GB")
   115→ * @param {boolean} isEpisodeRequest - Whether this is for a specific episode
   116→ * @returns {boolean} True if this looks like a pack
   117→ */
   118→function isPackFile(filename, size, isEpisodeRequest = false) {
   119→    if (!filename) return false;
   120→
   121→    const filenameLower = filename.toLowerCase();
   122→
   123→    // Check for pack keywords in filename
   124→    if (hasPackKeyword(filenameLower)) {
   125→        return true;
   126→    }
   127→
   128→    // For episode requests, check if size is suspiciously large (>3GB for a single episode)
   129→    if (isEpisodeRequest && size) {
   130→        const sizeMatch = size.match(/([0-9.]+)\s*(GB|TB)/i);
   131→        if (sizeMatch) {
   132→            const value = parseFloat(sizeMatch[1]);
   133→            const unit = sizeMatch[2].toUpperCase();
   134→            const sizeInGB = unit === 'TB' ? value * 1024 : value;
   135→
   136→            let maxEpisodeSizeGb = 5;
   137→            const resolution = getResolutionFromName(filename);
   138→            if (resolution === '2160p' || /\b(4k|uhd)\b/i.test(filenameLower)) {
   139→                maxEpisodeSizeGb = 25;
   140→            } else if (resolution === '1080p') {
   141→                maxEpisodeSizeGb = 8;
   142→            }
   143→
   144→            // Packs are typically much larger than single-episode files
   145→            if (sizeInGB > maxEpisodeSizeGb) {
   146→                console.log(`[HDHub4u] Skipping large file (${size}) > ${maxEpisodeSizeGb}GB as likely pack: ${filename}`);
   147→                return true;
   148→            }
   149→        }
   150→    }
   151→
   152→    return false;
   153→}

   163→function extractSeasonNumbersFromText(value) {
   164→    const seasons = new Set();
   165→    if (!value) return seasons;
   166→
   167→    const text = value.toLowerCase();
   168→
   169→    const rangeRegex = /\b(?:s|season[s]?)\s*0*(\d{1,2})\s*(?:-|–|to)\s*(?:s|season)?\s*0*(\d{1,2})\b/gi;
   170→    let match = null;
   171→    while ((match = rangeRegex.exec(text)) !== null) {
   172→        const start = parseInt(match[1], 10);
   173→        const end = parseInt(match[2], 10);
   174→        if (!Number.isFinite(start) || !Number.isFinite(end)) continue;
   175→        const min = Math.max(1, Math.min(start, end));
   176→        const max = Math.min(50, Math.max(start, end));
   177→        for (let season = min; season <= max; season += 1) {
   178→            seasons.add(season);
   179→        }
   180→    }
   181→
   182→    const singleRegex = /\b(?:season\s*0*(\d{1,2})|s0*(\d{1,2}))(?:\b|e\d+)/gi;
   183→    while ((match = singleRegex.exec(text)) !== null) {
   184→        const valueMatch = parseInt(match[1] || match[2], 10);
   185→        if (Number.isFinite(valueMatch)) {
   186→            seasons.add(valueMatch);
   187→        }
   188→    }
   189→
   190→    return seasons;
   191→}

   193→function scoreSeasonMatch(text, requestedSeason) {
   194→    if (!text || !requestedSeason) return 0;
   195→    const seasons = extractSeasonNumbersFromText(text);
   196→    if (seasons.size === 0) return 0;
   197→    return seasons.has(requestedSeason) ? 2 : -1;
   198→}

   218→function prioritizeLinks(downloadLinks, type, season, episode) {
   219→    const requestedSeason = season ? parseInt(season) : null;
   220→    const requestedEpisode = episode ? parseInt(episode) : null;
   221→
   222→    return downloadLinks
   223→        .map(link => {
   224→            let priority = 0;
   225→
   226→            // Prefer per-episode links for series
   227→            if (type === 'series') {
   228→                if (requestedSeason && link.season === requestedSeason) {
   229→                    priority += 30;
   230→                }
   231→                if (requestedEpisode && link.episode === requestedEpisode) {
   232→                    priority += 40;
   233→                }
   234→                if (!requestedEpisode && requestedSeason && link.label?.includes(`S${requestedSeason}`)) {
   235→                    priority += 20;
   236→                }
   237→            }
   238→
   239→            // Prefer higher resolution
   240→            const resolution = getResolutionFromName(link.label);
   241→            if (resolution === '2160p') priority += 25;
   242→            else if (resolution === '1080p') priority += 20;
   243→            else if (resolution === '720p') priority += 10;
   244→
   245→            // Prefer HEVC/265 encodes
   246→            if (/HEVC|H265|x265/i.test(link.label)) priority += 5;
   247→
   248→            // Slight preference for smaller sizes for faster extraction
   249→            if (link.size && /MB/i.test(link.size)) priority += 3;
   250→
   251→            return { ...link, priority };
   252→        })
   253→        .sort((a, b) => b.priority - a.priority);
   254→}

   256→async function processDownloadLink(link, index) {
   257→    try {
   258→        const results = await processExtractorLinkWithAwait(link.url, index + 1);
   259→        if (!results || results.length === 0) {
   260→            return [];
   261→        }
   262→
   263→        return results.map(result => ({
   264→            url: result.url,
   265→            name: result.name || 'HDHub4u',
   266→            quality: result.quality || getResolutionFromName(link.label),
   267→            size: link.size,
   268→            sourceLabel: link.label,
   269→            languages: link.languages?.length ? link.languages : detectLanguagesFromTitle(link.label),
   270→            resolverUrl: link.url
   271→        }));
   272→    } catch (error) {
   273→        console.error(`[HDHub4u] Failed to process link ${link.url}:`, error.message);
   274→        return [];
   275→    }
   276→}

   278→async function extractStreamingLinks(downloadLinks, type, season, episode) {
   279→    const prioritized = prioritizeLinks(downloadLinks, type, season, episode);
   280→    const limited = prioritized.slice(0, MAX_LINKS);
   281→
   282→    if (limited.length === 0) {
   283→        return [];
   284→    }
   285→
   286→    const concurrency = Math.min(MAX_THREAD_COUNT, limited.length);
   287→    console.log(`[HDHub4u] Extracting ${limited.length} links with concurrency ${concurrency}`);
   288→
   289→    const results = new Array(limited.length);
   290→    let cursor = 0;
   291→
   292→    const worker = async () => {
   293→        while (cursor < limited.length) {
   294→            const currentIndex = cursor++;
   295→            const link = limited[currentIndex];
   296→            results[currentIndex] = await processDownloadLink(link, currentIndex);
   297→        }
   298→    };
   299→
   300→    await Promise.all(Array.from({ length: concurrency }, worker));
   301→    return results.flat();
   302→}

   330→async function validateLinks(links) {
   331→    if (!links?.length) {
   332→        return [];
   333→    }
   334→
   335→    if (process.env.DISABLE_HDHUB4U_URL_VALIDATION === 'true') {
   336→        console.log('[HDHub4u] URL validation disabled via env, but enforcing 206 confirmation');
   337→    }
   338→
   339→    if (!SEEK_VALIDATION_ENABLED) {
   340→        console.log('[HDHub4u] Seek validation disabled via env override, forcing 206 confirmation for all links');
   341→    }
   342→
   343→    const trusted = [];
   344→    const otherLinks = [];
   345→    for (const link of links) {
   346→        if (!link.url) continue;
   347→        if (TRUSTED_HOSTS.some(host => link.url.includes(host))) {
   348→            trusted.push(link);
   349→        } else {
   350→            otherLinks.push(link);
   351→        }
   352→    }
   353→
   354→    const orderedLinks = [...trusted, ...otherLinks];
   355→    const validated = [];
   356→
   357→    for (let i = 0; i < orderedLinks.length; i += 4) {
   358→        const slice = orderedLinks.slice(i, i + 4);
   359→        const checks = await Promise.all(slice.map(async (link) => {
   360→            try {
   361→                const result = await validateSeekableUrl(link.url, { requirePartialContent: true });
   362→                if (!result.isValid) {
   363→                    console.log(`[HDHub4u] Dropped link (status ${result.statusCode || 'unknown'}) without confirmed 206 response: ${link.url}`);
   364→                    return null;
   365→                }
   366→                if (result.filename) {
   367→                    link.sourceLabel = `${result.filename} ${link.sourceLabel || ''}`.trim();
   368→                }
   369→                return link;
   370→            } catch (error) {
   371→                console.log(`[HDHub4u] Error validating ${link.url}: ${error.message}`);
   372→                return null;
   373→            }
   374→        }));
   375→
   376→        validated.push(...checks.filter(Boolean));
   377→    }
   378→
   379→    return validated;
   380→}

   382→function mapToStreams(links) {
   383→    const trustedDirectHosts = ['hubcloud', 'hubcdn', 'pixeldrain', 'r2.dev', 'workers.dev', 'googleusercontent.com'];
   384→
   385→    return links.map(link => {
   386→        let resolution = getResolutionFromName(link.sourceLabel);
   387→        if (resolution === 'other') {
   388→            resolution = getResolutionFromName(link.name);
   389→        }
   390→
   391→        let resolutionLabel = resolution;
   392→        if (resolution === '2160p') resolutionLabel = '4k';
   393→
   394→        const languages = link.languages?.length ? link.languages : detectLanguagesFromTitle(link.sourceLabel);
   395→        const languageFlags = renderLanguageFlags(languages);
   396→        let needsResolution = Boolean(link.resolverUrl);
   397→        const directUrl = encodeUrlForStreaming(link.url);
   398→
   399→        const urlLower = (link.url || '').toLowerCase();
   400→        const directIsTrusted = urlLower && trustedDirectHosts.some(host => urlLower.includes(host));
   401→
   402→        let streamUrl;
   403→        if (directIsTrusted) {
   404→            needsResolution = false;
   405→            streamUrl = encodeUrlForStreaming(link.url);
   406→        } else {
   407→            const resolverSource = needsResolution ? link.resolverUrl : link.url;
   408→            streamUrl = encodeUrlForStreaming(resolverSource || link.url);
   409→        }
   410→        const size = link.size || extractSizeFromLabel(link.sourceLabel || link.name);
   411→        const fileName = extractFileName(link.sourceLabel || link.name || '');
   412→        const behaviorHints = {
   413→            bingeGroup: 'hdhub4u-streams',
   414→            hdhub4uDirectUrl: directUrl
   415→        };
   416→        if (fileName) {
   417→            behaviorHints.fileName = fileName;
   418→        }
   419→
   420→        return {
   421→            name: `[HS+] Sootio\n${resolutionLabel}`,
   422→            title: `${normalizeLabel(link.sourceLabel || link.name)}${languageFlags}\n💾 ${size || 'N/A'} | HDHub4u`,
   423→            url: streamUrl,
   424→            size,
   425→            resolution,
   426→            needsResolution,
   427→            resolverFallbackUrl: directUrl,
   428→            behaviorHints
   429→        };
   430→    });
   431→}

   440→function filterEpisodeStreams(streams, season, episode) {
   441→    if (!season || !episode) return streams;
   442→    const requestedSeason = parseInt(season);
   443→    const requestedEpisode = parseInt(episode);
   444→
   445→    return streams.filter(stream => {
   446→        const title = stream.title || '';
   447→
   448→        // First, check for explicit SxxExx format which is most reliable
   449→        const sxxexxMatch = title.match(/S0*(\d+)\s*E0*(\d+)/i);
   450→        if (sxxexxMatch) {
   451→            const s = parseInt(sxxexxMatch[1]);
   452→            const e = parseInt(sxxexxMatch[2]);
   453→            return s === requestedSeason && e === requestedEpisode;
   454→        }
   455→
   456→        // Check for "Episode X" at the START of the title (before any "|" or "–" separators)
   457→        // This avoids matching button text like "Instant EPiSODE 1" at the end
   458→        const titleStart = title.split(/[|–-]/)[0];
   459→        const episodeMatch = titleStart.match(/Episode\s*0*(\d+)/i);
   460→        if (episodeMatch) {
   461→            const e = parseInt(episodeMatch[1]);
   462→            return e === requestedEpisode;
   463→        }
   464→
   465→        // Check for "EP X" or "Ep.X" format at the start
   466→        const epMatch = titleStart.match(/\bEP\.?\s*0*(\d+)/i);
   467→        if (epMatch) {
   468→            const e = parseInt(epMatch[1]);
   469→            return e === requestedEpisode;
   470→        }
   471→
   472→        // Fallback: no episode marker found, exclude by default for series
   473→        return false;
   474→    });
   475→}

   477→async function findBestMatch(searchResults, targetTitle) {
   478→    let bestMatch = null;
   479→    let bestScore = -Infinity;
   480→    const normalizedTarget = normalizeTitle(targetTitle);
   481→
   482→    for (const result of searchResults) {
   483→        const similarity = calculateSimilarity(normalizedTarget, result.slug);
   484→        const score = similarity - (result.score || 0);
   485→        if (score > bestScore) {
   486→            bestScore = score;
   487→            bestMatch = result;
   488→        }
   489→    }
   490→
   491→    return bestMatch;
   492→}

   494→export async function getHDHub4uStreams(imdbId, type, season = null, episode = null, config = {}, prefetchedMeta = null) {
   495→    // Signal not used in wrapper, but internal functions expect it - set to null
   496→    const signal = null;
   497→
   498→    try {
   499→        // Extract clientIp from config for per-IP rate limiting
   500→        const clientIp = config?.clientIp || null;
   501→
   502→        // Check FlareSolverr availability early (HDHub4u depends on it for HubCloud extraction)
   503→        // Only return error stream if IP is specifically rate-limited
   504→        if (clientIp && flaresolverrManager.isIpRateLimited(clientIp)) {
   505→            const remaining = flaresolverrManager.getIpRemainingRequests(clientIp);
   506→            console.warn(`[HDHub4u] Client IP ${clientIp} rate limited (${remaining} remaining)`);
   507→            return [createFlareSolverrErrorStream('rate_limited', { remaining })];
   508→        }
   509→
   510→        // Use pre-fetched metadata if available, otherwise fetch it (fallback for direct calls)
   511→        let cinemetaDetails = prefetchedMeta;
   512→        if (!cinemetaDetails) {
   513→            console.log(`[HDHub4u] No pre-fetched metadata, fetching from Cinemeta...`);
   514→            cinemetaDetails = await Cinemeta.getMeta(type, imdbId);
   515→        } else {
   516→            console.log(`[HDHub4u] Using pre-fetched Cinemeta metadata: "${cinemetaDetails.name}"`);
   517→        }
   518→
   519→        if (!cinemetaDetails) {
   520→            console.log('[HDHub4u] Cinemeta lookup failed');
   521→            return [];
   522→        }
   523→
   524→        const year = cinemetaDetails.year ? parseInt(String(cinemetaDetails.year).split('-')[0]) : null;
   525→
   526→        // Build query list: use alternative titles if available, otherwise use standard generation
   527→        let queries = [];
   528→        if (cinemetaDetails.alternativeTitles && cinemetaDetails.alternativeTitles.length > 0) {
   529→            console.log(`[HDHub4u] Using ${cinemetaDetails.alternativeTitles.length} alternative titles for search`);
   530→            queries = cinemetaDetails.alternativeTitles;
   531→        } else {
   532→            queries = generateAlternativeQueries(cinemetaDetails.name, cinemetaDetails.original_title);
   533→        }
   534→
   535→        let searchResults = [];
   536→        let usedYearFallback = false;
   537→        for (const query of queries) {
   538→            console.log(`[HDHub4u] Searching with query: "${query}"`);
   539→            const results = await searchHdHub4uPosts(query, 12);
   540→            if (results.length > 0) {
   541→                searchResults = results;
   542→                console.log(`[HDHub4u] Query "${query}" found ${results.length} results`);
   543→                break;
   544→            }
   545→        }
   546→
   547→        // Fallback: For recent movies/series with regional title mismatches
   548→        // This helps when Cinemeta has a different title than what's on HDHub4u
   549→        // (e.g., "Vampires of Vijay Nagar" vs "Thamma")
   550→        if (searchResults.length === 0 && year && year >= 2020) {
   551→            console.log(`[HDHub4u] Primary search failed, trying fallback with year ${year}`);
   552→
   553→            // For very recent movies, search by year with larger limit
   554→            // Fuse.js prioritizes shorter titles, so we need more results to find longer titles
   555→            const yearResults = await searchHdHub4uPosts(year.toString(), 100);
   556→            if (yearResults.length > 0) {
   557→                searchResults = yearResults;
   558→                usedYearFallback = true;
   559→                console.log(`[HDHub4u] Year-based fallback found ${yearResults.length} results`);
   560→            }
   561→        }
   562→
   563→        if (searchResults.length === 0) {
   564→            console.log('[HDHub4u] No search results found');
   565→            return [];
   566→        }
   567→
   568→        const isSeries = type === 'series' || type === 'tv';
   569→        if (isSeries && season) {
   570→            const requestedSeason = parseInt(season);
   571→            searchResults.sort((a, b) => {
   572→                const aScore = scoreSeasonMatch(a.slug, requestedSeason);
   573→                const bScore = scoreSeasonMatch(b.slug, requestedSeason);
   574→                if (aScore !== bScore) return bScore - aScore;
   575→                const aFuse = typeof a.score === 'number' ? a.score : 0;
   576→                const bFuse = typeof b.score === 'number' ? b.score : 0;
   577→                return aFuse - bFuse;
   578→            });
   579→        }
   580→
   581→        // Try multiple matches to find one with correct year/season
   582→        // For series with season requested, try more matches to find the correct season
   583→        // Check more results since Fuse.js prioritizes shorter titles
   584→        let content = null;
   585→        let matchIndex = 0;
   586→        const needsMultipleAttempts = usedYearFallback || (isSeries && season);
   587→        const maxAttempts = needsMultipleAttempts ? Math.min(80, searchResults.length) : Math.min(12, searchResults.length);
   588→
   589→        while (matchIndex < maxAttempts && !content) {
   590→            const bestMatch = searchResults[matchIndex];
   591→            if (!bestMatch) break;
   592→
   593→            // When using year fallback, skip very short generic titles (like "55", "Mrs", "G20")
   594→            // to prefer longer, more specific titles that might be regional variations
   595→            if (usedYearFallback) {
   596→                const titleWords = bestMatch.slug.split(/\s+/).filter(w => w.length > 0 && !/^\d{4}$/.test(w)); // Filter out year
   597→                const mainTitle = titleWords.filter(w => {
   598→                    const lowerWord = w.toLowerCase();
   599→                    return !['hindi', 'english', 'webrip', 'bluray', 'full', 'movie', 'series', 'hdtc', 'dubbed', 'dual', 'audio'].includes(lowerWord);
   600→                });
   601→
   602→                if (mainTitle.length > 0) {
   603→                    const titleText = mainTitle.join(' ');
   604→                    const titleLength = titleText.replace(/\s+/g, '').length;
   605→                    // Skip if title is very short (1-5 characters) unless we've checked many matches
   606→                    // This helps skip generic abbreviated titles like "55", "Mrs", "HAQ" in favor of
   607→                    // longer regional titles like "Thamma" that might appear later
   608→                    if (titleLength <= 5 && matchIndex < 60) {
   609→                        console.log(`[HDHub4u] Skipping short title "${titleText}" (${titleLength} chars) at position ${matchIndex + 1}, looking for more specific match`);
   610→                        matchIndex++;
   611→                        continue;
   612→                    }
   613→                }
   614→            }
   615→
   616→            console.log(`[HDHub4u] Trying match ${matchIndex + 1}: ${bestMatch.slug}`);
   617→            const candidateContent = await loadHdHub4uPost(bestMatch.url, signal);
   618→
   619→            if (!candidateContent || !candidateContent.downloadLinks?.length) {
   620→                console.log(`[HDHub4u] No download links found for ${bestMatch.url}`);
   621→                matchIndex++;
   622→                continue;
   623→            }
   624→
   625→            // Check year match when using fallback
   626→            if (usedYearFallback && year && candidateContent.year && Math.abs(candidateContent.year - year) > 1) {
   627→                console.log(`[HDHub4u] Year mismatch (${candidateContent.year} vs ${year}), trying next match`);
   628→                matchIndex++;
   629→                continue;
   630→            }
   631→
   632→            const normalizedTarget = normalizeTitle(cinemetaDetails.name);
   633→            const normalizedContentTitle = normalizeTitle(candidateContent.title || '');
   634→            const normalizedSlug = normalizeTitle(bestMatch.slug || '');
   635→
   636→            // Skip strict title matching for series when using year fallback (helps with regional title mismatches)
   637→            if (type !== 'movie' && normalizedTarget && !normalizedContentTitle.includes(normalizedTarget) && !usedYearFallback) {
   638→                console.log(`[HDHub4u] Skipping content due to title mismatch: "${candidateContent.title}" vs target "${normalizedTarget}"`);
   639→                matchIndex++;
   640→                continue;
   641→            }
   642→
   643→            // For movies, enforce a minimum title similarity to avoid wrong matches
   644→            if (type === 'movie' && normalizedTarget) {
   645→                const titleSimilarity = calculateSimilarity(normalizedTarget, normalizedContentTitle);
   646→                const minSimilarity = usedYearFallback ? 0.38 : 0.32;
   647→                const targetWords = normalizedTarget.split(' ').filter(Boolean);
   648→                const allowContainsFallback = targetWords.length > 1 || normalizedTarget.length >= 4;
   649→                const targetPattern = allowContainsFallback ? new RegExp(`\\b${escapeRegExp(normalizedTarget)}\\b`) : null;
   650→                const containsTarget = allowContainsFallback
   651→                    && (targetPattern.test(normalizedContentTitle) || targetPattern.test(normalizedSlug));
   652→
   653→                if (titleSimilarity < minSimilarity && !containsTarget) {
   654→                    console.log(`[HDHub4u] Skipping content due to low title similarity (${titleSimilarity.toFixed(3)} < ${minSimilarity}): "${candidateContent.title}"`);
   655→                    matchIndex++;
   656→                    continue;
   657→                }
   658→            }
   659→
   660→            // For movies, verify year match
   661→            if (type === 'movie' && year && candidateContent.year && Math.abs(candidateContent.year - year) > 1) {
   662→                console.log(`[HDHub4u] Year mismatch (${candidateContent.year} vs ${year})`);
   663→                matchIndex++;
   664→                continue;
   665→            }
   666→
   667→            // For series, verify that the requested season exists in the content
   668→            if ((type === 'series' || type === 'tv') && season) {
   669→                const requestedSeason = parseInt(season);
   670→                const contentTitle = (candidateContent.title || '').toLowerCase();
   671→
   672→                // Check if content title mentions a DIFFERENT season
   673→                const titleSeasons = extractSeasonNumbersFromText(contentTitle);
   674→                if (titleSeasons.size > 0 && !titleSeasons.has(requestedSeason)) {
   675→                    const listed = Array.from(titleSeasons).sort((a, b) => a - b).join(', ');
   676→                    console.log(`[HDHub4u] Season mismatch in title: found Season(s) ${listed}, requested Season ${requestedSeason} - "${candidateContent.title}"`);
   677→                    matchIndex++;
   678→                    continue;
   679→                }
   680→                const titleSeasonMatches = titleSeasons.size > 0 && titleSeasons.has(requestedSeason);
   681→
   682→                // Also check if any download links have the requested season
   683→                if (!titleSeasonMatches) {
   684→                    const hasRequestedSeason = candidateContent.downloadLinks?.some(link => {
   685→                        const linkSeason = link.season;
   686→                        if (linkSeason === requestedSeason) return true;
   687→
   688→                        // Check label for season markers
   689→                        const labelLower = (link.label || '').toLowerCase();
   690→                        const labelSeasonMatch = labelLower.match(/s0*(\d+)|season\s*(\d+)/i);
   691→                        if (labelSeasonMatch) {
   692→                            const labelSeason = parseInt(labelSeasonMatch[1] || labelSeasonMatch[2]);
   693→                            return labelSeason === requestedSeason;
   694→                        }
   695→
   696→                        return false;
   697→                    });
   698→
   699→                    if (!hasRequestedSeason && candidateContent.downloadLinks?.length > 0) {
   700→                        // Check if links have ANY season info - if they do and none match, skip
   701→                        const hasAnySeasonInfo = candidateContent.downloadLinks.some(link => {
   702→                            const labelLower = (link.label || '').toLowerCase();
   703→                            return link.season || /s\d+|season\s*\d+/i.test(labelLower);
   704→                        });
   705→
   706→                        if (hasAnySeasonInfo) {
   707→                            console.log(`[HDHub4u] Content has season info but not Season ${requestedSeason}, skipping "${candidateContent.title}"`);
   708→                            matchIndex++;
   709→                            continue;
   710→                        }
   711→                    }
   712→                }
   713→            }
   714→
   715→            // For series, skip candidates that only include different episodes
   716→            if ((type === 'series' || type === 'tv') && episode) {
   717→                const requestedEpisode = parseInt(episode);
   718→                const episodeLinks = (candidateContent.downloadLinks || []).filter(link => {
   719→                    if (link.episode !== null && link.episode !== undefined) {
   720→                        return true;
   721→                    }
   722→                    return getEpisodeFromLabel(link.label || '') !== null;
   723→                });
   724→
   725→                if (episodeLinks.length > 0) {
   726→                    const hasRequestedEpisode = episodeLinks.some(link => {
   727→                        if (link.episode !== null && link.episode !== undefined) {
   728→                            return link.episode === requestedEpisode;
   729→                        }
   730→                        return getEpisodeFromLabel(link.label || '') === requestedEpisode;
   731→                    });
   732→
   733→                    if (!hasRequestedEpisode) {
   734→                        const knownEpisodes = Array.from(new Set(episodeLinks.map(link => (
   735→                            link.episode ?? getEpisodeFromLabel(link.label || '')
   736→                        )).filter(Number.isFinite))).sort((a, b) => a - b);
   737→                        const listed = knownEpisodes.length > 0 ? knownEpisodes.join(', ') : 'unknown';
   738→                        console.log(`[HDHub4u] Episode mismatch: found Episode(s) ${listed}, requested Episode ${requestedEpisode} - "${candidateContent.title}"`);
   739→                        matchIndex++;
   740→                        continue;
   741→                    }
   742→                }
   743→            }
   744→
   745→            // Found a good match!
   746→            content = candidateContent;
   747→            console.log(`[HDHub4u] Using match: ${candidateContent.title}`);
   748→        }
   749→
   750→        if (!content) {
   751→            console.log(`[HDHub4u] No suitable content found after checking ${matchIndex} matches`);
   752→            return [];
   753→        }
   754→
   755→        // CHECK FOR LAZY-LOAD MODE
   756→        if (isLazyLoadEnabled()) {
   757→            console.log(`[HDHub4u] Lazy-load enabled: returning ${content.downloadLinks.length} preview streams without extraction/validation`);
   758→
   759→            // HDHub4u already has rich metadata in downloadLinks!
   760→            let linksToProcess = content.downloadLinks;
   761→
   762→            // Filter out "Instant" links - these are low-quality redirect links that often fail
   763→            const beforeInstantFilter = linksToProcess.length;
   764→            linksToProcess = linksToProcess.filter(link => {
   765→                if (link.isInstant === true) return false;
   766→                if (link.isInstant === false) return true;
   767→                const label = link.label || '';
   768→                if (!/\bInstant\b/i.test(label)) return true;
   769→                const host = getHostname(link.url);
   770→                if (host.includes('hubdrive') || host.includes('hubcloud')) return true;
   771→                return false;
   772→            });
   773→            if (linksToProcess.length < beforeInstantFilter) {
   774→                console.log(`[HDHub4u] Filtered out ${beforeInstantFilter - linksToProcess.length} "Instant" links`);
   775→            }
   776→
   777→            // PRE-FILTER by episode metadata for series
   778→            if ((type === 'series' || type === 'tv') && season && episode) {
   779→                const requestedSeason = parseInt(season);
   780→                const requestedEpisode = parseInt(episode);
   781→
   782→                // Filter links that have episode metadata matching the requested episode
   783→                const episodeFilteredLinks = linksToProcess.filter(link => {
   784→                    // If link has explicit episode number, it must match
   785→                    if (link.episode !== null && link.episode !== undefined) {
   786→                        return link.episode === requestedEpisode;
   787→                    }
   788→
   789→                    // Check label for episode markers at the START (before separators)
   790→                    const label = link.label || '';
   791→                    const labelStart = label.split(/[|–]/)[0];
   792→
   793→                    // Check for SxxExx format
   794→                    const sxxexxMatch = labelStart.match(/S0*(\d+)\s*E0*(\d+)/i);
   795→                    if (sxxexxMatch) {
   796→                        const e = parseInt(sxxexxMatch[2]);
   797→                        return e === requestedEpisode;
   798→                    }
   799→
   800→                    // Check for "Episode X" at the start
   801→                    const episodeMatch = labelStart.match(/Episode\s*0*(\d+)/i);
   802→                    if (episodeMatch) {
   803→                        const e = parseInt(episodeMatch[1]);
   804→                        return e === requestedEpisode;
   805→                    }
   806→
   807→                    // Check for "EP X" format
   808→                    const epMatch = labelStart.match(/\bEP\.?\s*0*(\d+)/i);
   809→                    if (epMatch) {
   810→                        const e = parseInt(epMatch[1]);
   811→                        return e === requestedEpisode;
   812→                    }
   813→
   814→                    // No episode marker - exclude for series
   815→                    return false;
   816→                });
   817→
   818→                if (episodeFilteredLinks.length > 0) {
   819→                    console.log(`[HDHub4u] Pre-filtered to ${episodeFilteredLinks.length} links for Episode ${requestedEpisode}`);
   820→                    linksToProcess = episodeFilteredLinks;
   821→                } else {
   822→                    console.log(`[HDHub4u] No links matched Episode ${requestedEpisode} by metadata, keeping all for title-based filtering`);
   823→                }
   824→            }
   825→
   826→            // Prioritize and limit the links
   827→            const prioritized = prioritizeLinks(linksToProcess, type, season, episode);
   828→            const limited = prioritized.slice(0, MAX_LINKS);
   829→
   830→            // Optional filename/size extraction (expensive; disabled by default in lazy-load mode)
   831→            let filenameMap = new Map();
   832→            if (EXTRACT_FILENAMES_IN_LAZY_MODE) {
   833→                try {
   834→                    const targets = limited.slice(0, EXTRACT_FILENAME_MAX_LINKS);
   835→                    if (targets.length > 0) {
   836→                        console.log(`[HDHub4u] Extracting file info from ${targets.length} drive links...`);
   837→                        filenameMap = await batchExtractFilenames(targets, {
   838→                            concurrency: EXTRACT_FILENAME_CONCURRENCY,
   839→                            timeoutMs: EXTRACT_FILENAME_TIMEOUT_MS
   840→                        });
   841→                    }
   842→                } catch (err) {
   843→                    console.log(`[HDHub4u] Filename extraction failed: ${err.message}`);
   844→                }
   845→            } else {
   846→                console.log('[HDHub4u] Skipping filename extraction in lazy-load mode');
   847→            }
   848→
   849→            // Determine if this is an episode request
   850→            const isEpisodeRequest = (type === 'series' || type === 'tv') && season && episode;
   851→
   852→            // Filter out packs and create preview streams
   853→            const previewStreams = [];
   854→            for (const link of limited) {
   855→                const extractedInfo = filenameMap.get(link.url);
   856→                const filename = extractedInfo?.filename || link.label || 'HDHub4u Stream';
   857→                const size = extractedInfo?.size || link.size;
   858→
   859→                // Skip packs for episode requests
   860→                if (isEpisodeRequest && isPackFile(filename, size, true)) {
   861→                    console.log(`[HDHub4u] Filtered out pack: ${filename} (${size})`);
   862→                    continue;
   863→                }
   864→
   865→                previewStreams.push(createPreviewStream({
   866→                    url: link.url,
   867→                    label: filename,
   868→                    provider: 'HDHub4u',
   869→                    size: size,
   870→                    languages: link.languages || []
   871→                }));
   872→            }
   873→
   874→            // If all links were filtered out as packs, return empty
   875→            if (previewStreams.length === 0) {
   876→                console.log('[HDHub4u] All links were filtered out (likely packs). Returning empty.');
   877→                return [];
   878→            }
   879→
   880→            // Format for Stremio
   881→            const streams = formatPreviewStreams(previewStreams, encodeUrlForStreaming, renderLanguageFlags);
   882→
   883→            // Apply additional episode filtering on formatted titles as fallback
   884→            if ((type === 'series' || type === 'tv') && season && episode) {
   885→                const episodeStreams = filterEpisodeStreams(streams, season, episode);
   886→                if (episodeStreams.length > 0) {
   887→                    console.log(`[HDHub4u] Returning ${episodeStreams.length} preview streams for S${season}E${episode} (lazy-load mode)`);
   888→                    return episodeStreams;
   889→                }
   890→
   891→                // No matching episode found - check if all links have DIFFERENT episode markers
   892→                // If so, the requested episode doesn't exist on this page
   893→                const requestedEpisode = parseInt(episode);
   894→                const allHaveDifferentEpisodes = streams.every(stream => {
   895→                    const title = stream.title || '';
   896→                    const titleStart = title.split(/[|–]/)[0];
   897→
   898→                    // Check for explicit episode markers
   899→                    const sxxexxMatch = titleStart.match(/S0*(\d+)\s*E0*(\d+)/i);
   900→                    if (sxxexxMatch) {
   901→                        return parseInt(sxxexxMatch[2]) !== requestedEpisode;
   902→                    }
   903→
   904→                    const episodeMatch = titleStart.match(/Episode\s*0*(\d+)/i);
   905→                    if (episodeMatch) {
   906→                        return parseInt(episodeMatch[1]) !== requestedEpisode;
   907→                    }
   908→
   909→                    const epMatch = titleStart.match(/\bEP\.?\s*0*(\d+)/i);
   910→                    if (epMatch) {
   911→                        return parseInt(epMatch[1]) !== requestedEpisode;
   912→                    }
   913→
   914→                    // Check for VOL patterns that indicate episode ranges (e.g., VOL-1 = Ep 1-4)
   915→                    const volMatch = title.match(/VOL[\s.-]*(\d+)/i);
   916→                    if (volMatch) {
   917→                        // VOL-1 typically contains episodes 1-4, VOL-2 contains 5-8, etc.
   918→                        const volNum = parseInt(volMatch[1]);
   919→                        const volStartEp = (volNum - 1) * 4 + 1;
   920→                        const volEndEp = volNum * 4;
   921→                        // If requested episode is outside this volume's range, it's a mismatch
   922→                        if (requestedEpisode < volStartEp || requestedEpisode > volEndEp) {
   923→                            return true;
   924→                        }
   925→                    }
   926→
   927→                    // No episode marker found - could potentially contain the episode
   928→                    return false;
   929→                });
   930→
   931→                if (allHaveDifferentEpisodes) {
   932→                    console.log(`[HDHub4u] Episode ${episode} not found - all links are for different episodes. Returning empty.`);
   933→                    return [];
   934→                }
   935→
   936→                // Some links don't have episode markers - return only those (potential packs)
   937→                const nonEpisodeStreams = streams.filter(stream => {
   938→                    const title = stream.title || '';
   939→                    const titleStart = title.split(/[|–]/)[0];
   940→
   941→                    // Exclude streams with explicit wrong episode markers
   942→                    const sxxexxMatch = titleStart.match(/S0*(\d+)\s*E0*(\d+)/i);
   943→                    if (sxxexxMatch && parseInt(sxxexxMatch[2]) !== requestedEpisode) {
   944→                        return false;
   945→                    }
   946→
   947→                    const episodeMatch = titleStart.match(/Episode\s*0*(\d+)/i);
   948→                    if (episodeMatch && parseInt(episodeMatch[1]) !== requestedEpisode) {
   949→                        return false;
   950→                    }
   951→
   952→                    const epMatch = titleStart.match(/\bEP\.?\s*0*(\d+)/i);
   953→                    if (epMatch && parseInt(epMatch[1]) !== requestedEpisode) {
   954→                        return false;
   955→                    }
   956→
   957→                    // Check VOL patterns
   958→                    const volMatch = title.match(/VOL[\s.-]*(\d+)/i);
   959→                    if (volMatch) {
   960→                        const volNum = parseInt(volMatch[1]);
   961→                        const volStartEp = (volNum - 1) * 4 + 1;
   962→                        const volEndEp = volNum * 4;
   963→                        if (requestedEpisode < volStartEp || requestedEpisode > volEndEp) {
   964→                            return false;
   965→                        }
   966→                    }
   967→
   968→                    return true;
   969→                });
   970→
   971→                if (nonEpisodeStreams.length > 0) {
   972→                    console.log(`[HDHub4u] Returning ${nonEpisodeStreams.length} non-episode-specific streams that may contain Episode ${episode}`);
   973→                    return nonEpisodeStreams;
   974→                }
   975→
   976→                console.log(`[HDHub4u] Episode ${episode} not available on this page. Returning empty.`);
   977→                return [];
   978→            }
   979→
   980→            console.log(`[HDHub4u] Returning ${streams.length} preview streams (lazy-load mode)`);
   981→            return streams;
   982→        }
   983→
   984→        // LEGACY MODE: Full extraction and validation (slow but thorough)
   985→        console.log(`[HDHub4u] Lazy-load disabled: extracting and validating all streams (legacy mode)`);
   986→        const streamingLinks = await extractStreamingLinks(content.downloadLinks, type, season, episode);
   987→        if (streamingLinks.length === 0) {
   988→            console.log('[HDHub4u] No streaming links after extraction');
   989→            return [];
   990→        }
   991→
   992→        const filtered = filterSuspicious(streamingLinks);
   993→        const unique = dedupeLinks(filtered);
   994→        const validated = await validateLinks(unique);
   995→        if (validated.length === 0) {
   996→            console.log('[HDHub4u] No validated links remained');
   997→            return [];
   998→        }
   999→
  1000→        // ALWAYS filter out googleusercontent.com - user requested to NEVER return these
  1001→        const googleUserContentCount = validated.filter(link => link.url?.includes('googleusercontent.com')).length;
  1002→        const finalValidated = validated.filter(link => !link.url?.includes('googleusercontent.com'));
  1003→
  1004→        if (googleUserContentCount > 0) {
  1005→            console.log(`[HDHub4u] Filtered out ${googleUserContentCount} googleusercontent.com link(s), keeping ${finalValidated.length} other links`);
  1006→        }
  1007→
  1008→        if (finalValidated.length === 0) {
  1009→            console.log('[HDHub4u] No links remaining after filtering googleusercontent.com');
  1010→            return [];
  1011→        }
  1012→
  1013→        let streams = mapToStreams(finalValidated);
  1014→        streams.sort((a, b) => {
  1015→            const priority = { '2160p': 4, '1080p': 3, '720p': 2, '480p': 1, other: 0 };
  1016→            const resDiff = (priority[b.resolution] || 0) - (priority[a.resolution] || 0);
  1017→            if (resDiff !== 0) return resDiff;
  1018→
  1019→            // If resolutions are the same, sort by size (larger first)
  1020→            // Convert sizes to MB for proper comparison
  1021→            const getSizeInMB = (sizeStr) => {
  1022→                if (!sizeStr) return 0;
  1023→                const match = sizeStr.match(/([0-9.]+)\s*(GB|MB|TB)/i);
  1024→                if (!match) return 0;
  1025→                const value = parseFloat(match[1]);
  1026→                const unit = match[2].toUpperCase();
  1027→                if (unit === 'TB') return value * 1024 * 1024;
  1028→                if (unit === 'GB') return value * 1024;
  1029→                if (unit === 'MB') return value;
  1030→                return 0;
  1031→            };
  1032→            const sizeA = getSizeInMB(a.size);
  1033→            const sizeB = getSizeInMB(b.size);
  1034→            return sizeB - sizeA;
  1035→        });
  1036→
  1037→        if ((type === 'series' || type === 'tv') && season && episode) {
  1038→            const episodeStreams = filterEpisodeStreams(streams, season, episode);
  1039→            if (episodeStreams.length > 0) {
  1040→                streams = episodeStreams;
  1041→            }
  1042→        }
  1043→
  1044→        console.log(`[HDHub4u] Returning ${streams.length} streams`);
  1045→        return streams;
  1046→    } catch (error) {
  1047→        console.error('[HDHub4u] Error getting streams:', error.message);
  1048→        return [];
  1049→    }
  1050→}
