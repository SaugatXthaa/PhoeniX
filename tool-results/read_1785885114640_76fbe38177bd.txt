    51→function createProxyAgent(proxyUrl) {
    52→    if (!proxyUrl) return null;
    53→    try {
    54→        const lower = proxyUrl.toLowerCase();
    55→        if (lower.startsWith('socks4://') || lower.startsWith('socks5://') || lower.startsWith('socks5h://')) {
    56→            return new SocksProxyAgent(proxyUrl);
    57→        }
    58→        if (lower.startsWith('http://') || lower.startsWith('https://')) {
    59→            return new HttpsProxyAgent(proxyUrl);
    60→        }
    61→    } catch (error) {
    62→        console.warn(`[${PROVIDER}] Failed to create proxy agent ${proxyUrl}: ${error.message}`);
    63→    }
    64→    return null;
    65→}

    69→async function makeCineDozeRequest(url, options = {}) {
    70→    const {
    71→        parseHTML = false,
    72→        timeout = 5000,
    73→        headers = {},
    74→        disableProxy = false
    75→    } = options;
    76→
    77→    if (!CINEDOZE_PROXY_AGENT || disableProxy) {
    78→        return makeRequest(url, options);
    79→    }
    80→
    81→    const response = await axios.get(url, {
    82→        timeout,
    83→        headers: {
    84→            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36',
    85→            ...headers
    86→        },
    87→        maxRedirects: 5,
    88→        validateStatus: () => true,
    89→        responseType: 'text',
    90→        transformResponse: [data => data],
    91→        httpAgent: CINEDOZE_PROXY_AGENT,
    92→        httpsAgent: CINEDOZE_PROXY_AGENT,
    93→        proxy: false
    94→    });
    95→
    96→    const body = typeof response.data === 'string' ? response.data : String(response.data ?? '');
    97→    return {
    98→        statusCode: response.status,
    99→        headers: response.headers || {},
   100→        body,
   101→        document: parseHTML ? cheerio.load(body) : null,
   102→        url: response.request?.res?.responseUrl || url
   103→    };
   104→}

   109→function isCloudflareChallenge(body = '', statusCode = null) {
   110→    const lower = (body || '').toLowerCase();
   111→
   112→    const strongSignals = [
   113→        lower.includes('cf-mitigated'),
   114→        lower.includes('just a moment'),
   115→        lower.includes('checking your browser'),
   116→        lower.includes('attention required! | cloudflare'),
   117→        lower.includes('cf-turnstile'),
   118→        lower.includes('verify_turnstile'),
   119→        lower.includes('enable javascript and cookies to continue')
   120→    ];
   121→    if (strongSignals.some(Boolean)) {
   122→        return true;
   123→    }
   124→
   125→    // Soft markers can appear on real pages; require multiple markers (or 403/429) to classify as challenge.
   126→    const softSignals = {
   127→        cfChl: lower.includes('cf_chl'),
   128→        challengePlatform: lower.includes('challenge-platform') || lower.includes('/cdn-cgi/challenge-platform'),
   129→        cfBrowserVerification: lower.includes('cf-browser-verification'),
   130→        securityCheck: lower.includes('security check') && lower.includes('cloudflare')
   131→    };
   132→    const softCount = Object.values(softSignals).filter(Boolean).length;
   133→
   134→    if (softCount >= 2) {
   135→        return true;
   136→    }
   137→
   138→    if ((statusCode === 403 || statusCode === 429) && softCount > 0) {
   139→        return true;
   140→    }
   141→
   142→    return false;
   143→}

   149→    return true;

   152→function getCloudflareMarkers(body = '') {
   153→    const lower = (body || '').toLowerCase();
   154→    const markers = [];
   155→    if (lower.includes('cf-mitigated')) markers.push('cf-mitigated');
   156→    if (lower.includes('just a moment')) markers.push('just-a-moment');
   157→    if (lower.includes('cf_chl')) markers.push('cf_chl');
   158→    if (lower.includes('challenge-platform')) markers.push('challenge-platform');
   159→    if (lower.includes('cf-turnstile')) markers.push('cf-turnstile');
   160→    if (lower.includes('verify_turnstile')) markers.push('verify_turnstile');
   161→    if (lower.includes('security check')) markers.push('security-check');
   162→    if (lower.includes('cloudflare')) markers.push('cloudflare');
   163→    return markers;
   164→}

   266→async function searchCineDoze(query) {
   267→    // CineDoze search breaks when query contains colons - strip them
   268→    const cleanQuery = query.replace(/:/g, '').replace(/\s+/g, ' ').trim();
   269→
   270→    // Check in-memory cache first
   271→    const cacheKey = cleanQuery.toLowerCase();

   273→    if (cached && Date.now() - cached.ts < SEARCH_CACHE_TTL) {
   274→        console.log(`[${PROVIDER}] Search cache hit (memory) for "${query}"`);
   275→        return cached.data;
   276→    }
   277→
   278→    const url = `${BASE_URL}/search/${encodeURIComponent(cleanQuery)}/`;
   279→    try {
   280→        let response = await makeCineDozeRequest(url, { parseHTML: true, timeout: 8000 });
   281→        let $ = response.document;
   282→        let body = response.body || '';
   283→
   284→        // Check if we're blocked by Cloudflare

   287→            const flareResponse = await fetchWithFlareSolverr(url);
   288→
   289→            if (flareResponse && flareResponse.document) {
   290→                $ = flareResponse.document;
   291→                body = flareResponse.body || '';
   292→                console.log(`[${PROVIDER}] FlareSolverr bypass successful for search "${query}"`);
   293→            } else {
   294→                console.log(`[${PROVIDER}] FlareSolverr bypass failed for search "${query}"`);
   295→                return [];
   296→            }
   297→        }
   298→
   299→        const results = [];
   300→
   301→        $('article').each((_, article) => {
   302→            const link =
   303→                $(article).find('a[href*="/movies/"], a[href*="/tvshows/"]').first().attr('href');

   312→                results.push({ title, url: absolute });

   335→        return results;
   336→    } catch (err) {
   337→        console.log(`[${PROVIDER}] Search failed for "${query}": ${err.message}`);

   400→function extractHostLinks(html, baseUrl) {
   401→    const hostLinks = [];
   402→    const regex = /https?:\/\/[^\s"'<>]+/gi;
   403→    const matches = [...(html || '').matchAll(regex)];
   404→    const seen = new Set();
   405→
   406→    for (const m of matches) {
   407→        const href = toAbsolute(m[0], baseUrl);
   408→        if (!href || seen.has(href)) continue;
   409→        const lower = href.toLowerCase();
   410→        if (
   411→            lower.includes('hubdrive') ||
   412→            lower.includes('hubcloud') ||
   413→            lower.includes('hubcdn') ||
   414→            lower.includes('gdflix') ||
   415→            lower.includes('filepress') ||
   416→            lower.includes('pixeldrain') ||
   417→            lower.includes('filesdl')
   418→        ) {
   419→            hostLinks.push(href);
   420→            seen.add(href);
   421→        }
   422→    }
   423→
   424→    return hostLinks;
   425→}

   512→function buildStream(result, context) {
   513→    if (!result?.url) return null;
   514→
   515→    const labelBase = cleanText(result.title || result.name || context.quality || '');
   516→    const size = result.size || context.size || parseSizeFromText(labelBase) || parseSizeFromText(context.quality) || null;
   517→    const qualityLabel = getResolutionFromName(labelBase || result.name || context.quality || '') || 'HTTP';
   518→    const resLabel = qualityLabel === '2160p' ? '4k' : qualityLabel;
   519→    const languages = Array.from(
   520→        new Set([
   521→            ...(context.languages || []),
   522→            ...detectLanguagesFromTitle(labelBase),
   523→            ...detectLanguagesFromTitle(context.quality || ''),
   524→            ...detectLanguagesFromTitle(result.title || '')
   525→        ].filter(Boolean))
   526→    );
   527→    const languageFlags = renderLanguageFlags(languages);
   528→    const sizeInfo = size ? `\n💾 ${size} | ${PROVIDER}` : `\n${PROVIDER}`;
   529→    const title = `${labelBase || context.quality || 'Download'}${languageFlags}${sizeInfo}`;
   530→    const fileName = extractFileName(result.title || result.name || '');
   531→    const behaviorHints = {
   532→        bingeGroup: 'cinedoze-http'
   533→    };
   534→    if (fileName) {
   535→        behaviorHints.fileName = fileName;
   536→    }
   537→
   538→    return {
   539→        name: `[HS+] Sootio\n${resLabel}`,
   540→        title,
   541→        url: encodeUrlForStreaming(result.url),
   542→        size,
   543→        resolution: resLabel,
   544→        languages,
   545→        behaviorHints,
   546→        httpProvider: PROVIDER
   547→    };
   548→}

   554→async function resolveHostLinks(hostLinks, context) {
   555→    const hubLinks = hostLinks.filter(h => /hubdrive|hubcloud|hubcdn/.test(h));
   556→    const gdflixLinks = hostLinks.filter(h => /gdflix/.test(h));
   557→
   558→    // 1) Try hubdrive/hubcloud first
   559→    for (const link of hubLinks) {
   560→        try {
   561→            const extracted = await processExtractorLinkWithAwait(link, 1);
   562→            if (extracted && extracted.length > 0) {
   563→                const streams = extracted.map(r => buildStream(r, context)).filter(Boolean);
   564→                if (streams.length > 0) return streams;
   565→            }
   566→        } catch (err) {
   567→            console.log(`[${PROVIDER}] Hub link failed ${link}: ${err.message}`);
   568→        }
   569→    }
   570→
   571→    // 2) Fallback to gdflix -> only pixeldrain results
   572→    for (const link of gdflixLinks) {
   573→        try {
   574→            const extracted = await processExtractorLinkWithAwait(link, 2);
   575→            const pixelOnly = filterPixeldrainOnly(extracted);
   576→            if (pixelOnly && pixelOnly.length > 0) {
   577→                const streams = pixelOnly.map(r => buildStream(r, context)).filter(Boolean);
   578→                if (streams.length > 0) return streams;
   579→            }
   580→        } catch (err) {
   581→            console.log(`[${PROVIDER}] GDFlix fallback failed ${link}: ${err.message}`);
   582→        }
   583→    }
   584→
   585→    return [];
   586→}

   588→export async function getCineDozeStreams(tmdbId, type, season = null, episode = null, config = {}, prefetchedMeta = null) {
   589→    try {
   590→        console.log(`[${PROVIDER}] Starting search for ${tmdbId} (${type}${season ? ` S${season}` : ''}${episode ? `E${episode}` : ''})`);
   591→
   592→        // Use pre-fetched metadata if available, otherwise fetch it (fallback for direct calls)
   593→        let meta = prefetchedMeta;
   594→        if (!meta) {
   595→            console.log(`[${PROVIDER}] No pre-fetched metadata, fetching from Cinemeta...`);
   596→            meta = await Cinemeta.getMeta(type, tmdbId);
   597→        } else {
   598→            console.log(`[${PROVIDER}] Using pre-fetched Cinemeta metadata: "${meta.name}"`);
   599→        }
   600→
   601→        if (!meta?.name) {
   602→            console.log(`[${PROVIDER}] Missing metadata for ${tmdbId}`);
   603→            return [];
   604→        }
   605→
   606→        const queries = Array.from(new Set([
   607→            meta.name,
   608→            removeYear(meta.name),
   609→            ...(meta.alternativeTitles || []),
   610→            ...generateAlternativeQueries(meta.name, meta.original_title)
   611→        ].filter(Boolean)));
   612→
   613→        // Run searches in parallel for speed
   614→        console.log(`[${PROVIDER}] Searching with ${queries.length} queries in parallel:`, queries);
   615→        const searchPromises = queries.map(query => searchCineDoze(query).then(results => ({ query, results })));
   616→        const searchResponses = await Promise.all(searchPromises);
   617→
   618→        const searchResults = [];
   619→        for (const { query, results } of searchResponses) {
   620→            console.log(`[${PROVIDER}] Query "${query}" returned ${results.length} results`);
   621→            searchResults.push(...results);
   622→        }
   623→
   624→        if (searchResults.length === 0) {
   625→            console.log(`[${PROVIDER}] No search results for ${meta.name}`);
   626→            return [];
   627→        }
   628→
   629→        console.log(`[${PROVIDER}] Total ${searchResults.length} results before dedup/scoring`);
   630→        const best = getSortedMatches(searchResults, meta.name)[0];
   631→        if (!best?.url) {
   632→            console.log(`[${PROVIDER}] No suitable match for ${meta.name}`);
   633→            return [];
   634→        }
   635→
   636→        console.log(`[${PROVIDER}] Selected match: ${best.title} -> ${best.url}`);
   637→        const downloadEntries = await loadCineDozePage(best.url);
   638→        if (downloadEntries.length === 0) {
   639→            console.log(`[${PROVIDER}] No download entries found`);
   640→            return [];
   641→        }
   642→
   643→        // Check if lazy-load mode is enabled (default: true for faster initial response)
   644→        const useLazyLoad = isLazyLoadEnabled();
   645→
   646→        if (useLazyLoad) {
   647→            // Lazy-load mode: return preview streams without HubCloud extraction
   648→            console.log(`[${PROVIDER}] Lazy-load: returning ${downloadEntries.length} preview streams without extraction`);
   649→
   650→            const movieName = meta.name || 'Unknown';
   651→            const movieYear = meta.year || '';
   652→
   653→            const previewStreams = downloadEntries.map(entry => {
   654→                const label = `${movieName}${movieYear ? ` (${movieYear})` : ''} - ${entry.quality || 'Download'}`;
   655→                const languages = entry.languages?.length
   656→                    ? entry.languages
   657→                    : detectLanguagesFromTitle(entry.quality || '');
   658→                return createPreviewStream({
   659→                    url: entry.url,
   660→                    label,
   661→                    provider: PROVIDER,
   662→                    size: entry.size,
   663→                    languages
   664→                });
   665→            });
   666→
   667→            const streams = formatPreviewStreams(previewStreams, encodeUrlForStreaming, renderLanguageFlags);
   668→            console.log(`[${PROVIDER}] Lazy-load: returning ${streams.length} preview streams`);
   669→            return streams;
   670→        }
   671→
   672→        // Full extraction mode (when lazy-load is disabled)
   673→        const streamPromises = downloadEntries.map(async (entry) => {
   674→            const hostLinks = await expandCineDozeLink(entry.url);
   675→            if (!hostLinks.length) return [];
   676→            return resolveHostLinks(hostLinks, entry);
   677→        });
   678→
   679→        const resolved = (await Promise.all(streamPromises)).flat().filter(Boolean);
   680→
   681→        // Deduplicate by URL
   682→        const seen = new Set();
   683→        const streams = [];
   684→        for (const stream of resolved) {
   685→            if (!stream.url || seen.has(stream.url)) continue;
   686→            seen.add(stream.url);
   687→            streams.push(stream);
   688→        }
   689→
   690→        console.log(`[${PROVIDER}] Returning ${streams.length} streams`);
   691→        return streams;
   692→    } catch (err) {
   693→        console.error(`[${PROVIDER}] Unexpected error: ${err.message}`);
   694→        return [];
   695→    }
   696→}
