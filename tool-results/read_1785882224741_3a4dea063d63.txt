   174→function normalizeSolverProxyUrl(proxyUrl = '') {
   175→    const raw = String(proxyUrl || '').trim();
   176→    if (!raw) return '';
   177→    // Byparr accepts socks5, while many configs here use socks5h for remote DNS.
   178→    // Keep auth/host/port intact and only normalize the scheme when needed.
   179→    if (raw.toLowerCase().startsWith('socks5h://')) {
   180→        return `socks5://${raw.slice('socks5h://'.length)}`;
   181→    }
   182→    return raw;
   183→}

   185→function parseBrowserProxyConfig(proxyUrl = '') {
   186→    const normalized = normalizeSolverProxyUrl(proxyUrl);
   187→    if (!normalized) {
   188→        return { proxyServer: null, username: null, password: null };
   189→    }
   190→
   191→    try {
   192→        const parsed = new URL(normalized);
   193→        const protocol = parsed.protocol.replace(/:$/, '');
   194→        if (!['socks5', 'http', 'https'].includes(protocol)) {
   195→            return { proxyServer: null, username: null, password: null };
   196→        }
   197→
   198→        const authUser = decodeURIComponent(parsed.username || '');
   199→        const authPass = decodeURIComponent(parsed.password || '');
   200→        const host = parsed.port ? `${parsed.hostname}:${parsed.port}` : parsed.hostname;
   201→        const proxyServer = `${protocol}://${host}`;
   202→
   203→        return {
   204→            proxyServer,
   205→            username: authUser || null,
   206→            password: authPass || null
   207→        };
   208→    } catch {
   209→        return { proxyServer: null, username: null, password: null };
   210→    }
   211→}

   371→function mergeCookieHeader(existing, setCookieHeader) {
   372→    const cookieMap = new Map();
   373→    if (existing) {
   374→        existing.split(';').forEach(cookie => {
   375→            const [name, ...rest] = cookie.trim().split('=');
   376→            if (!name || rest.length === 0) return;
   377→            cookieMap.set(name, rest.join('='));
   378→        });
   379→    }
   380→    extractCookies(setCookieHeader).forEach(cookie => {
   381→        const [name, ...rest] = cookie.split('=');
   382→        if (!name || rest.length === 0) return;
   383→        cookieMap.set(name, rest.join('='));
   384→    });
   385→    return Array.from(cookieMap.entries())
   386→        .map(([name, value]) => `${name}=${value}`)
   387→        .join('; ');
   388→}

   390→function parseCookieHeader(cookieHeader = '', domain = null) {
   391→    if (!cookieHeader) return [];
   392→    return cookieHeader.split(';').map(part => part.trim()).map((part) => {
   393→        if (!part) return null;
   394→        const [name, ...rest] = part.split('=');
   395→        if (!name || rest.length === 0) return null;
   396→        return {
   397→            name,
   398→            value: rest.join('='),
   399→            ...(domain ? { domain } : {})
   400→        };
   401→    }).filter(Boolean);
   402→}

   437→/**

   439→ * FlareSolverr returns cookies as array: [{ name, value, domain, ... }]

   441→ * @param {Array} cookies - Array of cookie objects from FlareSolverr
   442→ * @param {string} userAgent - The user agent used (cookies are tied to UA)
   443→ */
   444→function cacheCfCookies(domain, cookies, userAgent) {
   445→    if (!domain || !cookies || !Array.isArray(cookies) || cookies.length === 0) return;
   446→
   447→    // Cache all cookies (not only cf_*). Some providers require session cookies
   448→    // across archive -> getlink -> redirect hops even after the Cloudflare challenge.
   449→    const cookieString = cookies
   450→        .filter(c => c?.name && c?.value !== undefined)
   451→        .map(c => `${c.name}=${c.value}`)
   452→        .join('; ');
   453→    if (!cookieString) return;
   454→
   455→    const cacheEntry = {
   456→        cookies: cookieString,
   457→        userAgent: userAgent || OUO_USER_AGENT,
   458→        ts: Date.now()
   459→    };
   460→
   461→    // Cache for the domain and all related domains
   462→    const domains = getRelatedDomains(domain);
   463→    for (const d of domains) {
   464→        cfCookieCache.set(d, cacheEntry);
   465→    }
   466→
   467→    console.log(`[HTTP-RESOLVE] Cached cookies for ${domains.join(', ')}: ${cookies.map(c => c.name).join(', ')}`);
   468→}

   470→function extractRedirectCandidates(body = '', document = null, baseUrl = '') {
   471→    const candidates = [];
   472→
   473→    if (document) {
   474→        const refresh = document('meta[http-equiv=\"refresh\"]').attr('content') || '';
   475→        const refreshMatch = refresh.match(/url=([^;]+)/i);
   476→        if (refreshMatch?.[1]) {
   477→            const resolved = normalizeAbsoluteUrl(refreshMatch[1].trim(), baseUrl);
   478→            if (resolved) candidates.push(resolved);
   479→        }
   480→
   481→        document('a[href]').each((_, el) => {
   482→            const href = document(el).attr('href');
   483→            const resolved = normalizeAbsoluteUrl(href, baseUrl);
   484→            if (resolved) candidates.push(resolved);
   485→        });
   486→    }
   487→
   488→    const scriptMatches = body.match(/location\.(?:href|replace|assign)\s*(?:\(\s*)?['"]([^'"]+)['"]\s*\)?/i);
   489→    if (scriptMatches?.[1]) {
   490→        const resolved = normalizeAbsoluteUrl(scriptMatches[1], baseUrl);
   491→        if (resolved) candidates.push(resolved);
   492→    }
   493→
   494→    const windowOpenMatch = body.match(/window\.open\(\s*['"]([^'"]+)['"]/i);
   495→    if (windowOpenMatch?.[1]) {
   496→        const resolved = normalizeAbsoluteUrl(windowOpenMatch[1], baseUrl);
   497→        if (resolved) candidates.push(resolved);
   498→    }
   499→
   500→    const urlMatches = body.match(/https?:\/\/[^\s"'<>]+/gi) || [];
   501→    urlMatches.forEach(match => {
   502→        const resolved = normalizeAbsoluteUrl(match, baseUrl);
   503→        if (resolved) candidates.push(resolved);
   504→    });
   505→
   506→    const base64Matches = body.match(/[A-Za-z0-9+/=]{40,}/g) || [];
   507→    base64Matches.forEach(raw => {
   508→        const decoded = tryDecodeBase64(raw);
   509→        if (decoded && decoded.startsWith('http')) {

   513→    });
   514→

   522→function isHomepageUrl(url) {
   523→    try {

   532→        if (!path || path === '' || path === '/home' || path === '/index' || path === '/index.html') {
   533→            return true;
   534→        }
   535→        return false;
   536→    } catch {
   537→        return false;
   538→    }

   586→function decodeBase64UrlSuffix(value = '') {
   587→    const raw = String(value || '').trim();
   588→    if (!raw || raw.length < 16) return null;
   589→
   590→    for (let i = 0; i < raw.length; i += 1) {
   591→        const suffix = raw.slice(i);
   592→        if (!/^[A-Za-z0-9+/=]+$/.test(suffix)) continue;
   593→        if (suffix.length % 4 !== 0) continue;
   594→        const decoded = tryDecodeBase64(suffix);
   595→        if (decoded && /^https?:\/\//i.test(decoded)) {
   596→            return decoded.trim();
   597→        }
   598→    }
   599→
   600→    return null;
   601→}

   668→function parseStreamHints(rawUrl) {
   669→    if (!rawUrl) {
   670→        return { baseUrl: rawUrl, hints: {} };
   671→    }
   672→
   673→    let baseUrl = rawUrl;
   674→    let hashParams = new URLSearchParams();
   675→    const hashIndex = rawUrl.indexOf('#');
   676→    if (hashIndex >= 0) {
   677→        baseUrl = rawUrl.slice(0, hashIndex);
   678→        hashParams = new URLSearchParams(rawUrl.slice(hashIndex + 1));
   679→    }
   680→
   681→    let queryEpisodeHint = null;
   682→    let queryResolutionHint = null;
   683→    let queryHostHint = null;
   684→    let queryPasswordHints = [];

   691→        queryPasswordHints = parsed.searchParams.getAll('sootio_pwd');

   694→        parsed.searchParams.delete('sootio_ep');
   695→        parsed.searchParams.delete('sootio_res');
   696→        parsed.searchParams.delete('sootio_host');
   697→        parsed.searchParams.delete('sootio_pwd');
   698→        baseUrl = parsed.toString();
   699→    } catch {
   700→        // Leave baseUrl untouched if it is not a valid URL.
   701→    }
   702→
   703→    return {
   704→        baseUrl,
   705→        hints: {
   706→            episode: hashParams.get('ep') || queryEpisodeHint || null,
   707→            resolution: hashParams.get('res') || queryResolutionHint || null,
   708→            host: hashParams.get('host') || queryHostHint || null,

   786→        } catch { /* ignore */ }

   841→function rankProviderArchiveCandidate(candidate, baseUrl, hints = {}) {
   842→    if (!candidate) return -1;
   843→    let score = 0;
   844→    const lower = candidate.toLowerCase();
   845→    const isUhdSid = /\?sid=/i.test(candidate) && UHDMOVIES_SID_HOSTS.some(h => lower.includes(h));
   846→    const isGdriveProRedirect = lower.includes('gdrivepro.') && /\/r\.php\?id=/i.test(lower);
   847→    const isUrlFlixGet = lower.includes('urlflix.xyz') && /\/gets?\//i.test(lower);
   848→
   849→    if (isProviderArchiveGetLinkUrl(candidate)) score += 500;
   850→    if (isUhdSid) score += 480;
   851→    if (isProviderArchiveWrapperUrl(candidate)) score += 400;
   852→    if (PROVIDER_ARCHIVE_TARGET_HOST_HINTS.some(h => lower.includes(h))) score += 300;
   853→    if (isUrlFlixGet) score += 220;
   854→    if (isGdriveProRedirect) score += 180;
   855→    if (DIRECT_HOST_HINTS.some(h => lower.includes(h))) score += 250;
   856→    if (hints?.host && lower.includes(String(hints.host).toLowerCase())) score += 50;
   857→
   858→    try {
   859→        const candidateHost = new URL(candidate).hostname.toLowerCase();
   860→        const baseHost = new URL(baseUrl).hostname.toLowerCase();
   861→        if (candidateHost !== baseHost) score += 40;
   862→    } catch {
   863→        // ignore parse failures
   864→    }
   865→
   866→    if (isAssetUrl(candidate) || isHomepageUrl(candidate)) score -= 1000;
   867→    return score;
   868→}

   878→    if (PROVIDER_ARCHIVE_TARGET_HOST_HINTS.some(h => lower.includes(h))) return true;

   939→async function resolveProviderArchiveGetLink(getLinkUrl, referer = null, hints = {}) {

   968→    }

   970→    if (response?.url && response.url !== getLinkUrl) {
   971→        return response.url;
   972→    }
   973→
   974→    return null;
   975→}

   977→async function resolveProviderArchiveWrapper(archiveUrl, hints = {}, depth = 0, visited = new Set()) {
   978→    if (!archiveUrl) return null;
   979→    if (depth > 3) return null;
   980→    if (visited.has(archiveUrl)) return null;
   981→    visited.add(archiveUrl);
   982→
   983→    console.log(`[HTTP-RESOLVE] Expanding provider archive wrapper (depth ${depth + 1}): ${archiveUrl.substring(0, 100)}...`);
   984→
   985→    const response = await fetchWithCloudflare(archiveUrl, {
   986→        timeout: 15000,
   987→        preferFlareSolverr: true,
   988→        headers: {
   989→            'User-Agent': OUO_USER_AGENT,
   990→            'Referer': archiveUrl,
   991→            'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,*/*;q=0.8',
   992→            'Accept-Language': 'en-US,en;q=0.5',
   993→            'Connection': 'keep-alive',
   994→            'Upgrade-Insecure-Requests': '1'
   995→        }
   996→    });
   997→
   998→    if (response?.url && response.url !== archiveUrl && !isProviderArchiveDomain((() => {
   999→        try { return new URL(response.url).hostname; } catch { return ''; }
  1000→    })())) {
  1001→        return response.url;
  1002→    }
  1003→
  1004→    // Provider archive pages (AnimeFlix / Episodes.ModPro) often contain episode-labeled links.
  1005→    // If we have an episode hint, try exact matching episode entries first.
  1006→    try {
  1007→        const episodeCandidates = collectProviderArchiveEpisodeLinks(response, archiveUrl, hints);
  1008→        if (episodeCandidates.length) {
  1009→            console.log(`[HTTP-RESOLVE] Provider episode hint ${hints.episode}: trying ${episodeCandidates.length} candidate(s)`);
  1010→            for (const candidate of episodeCandidates) {
  1011→                if (visited.has(candidate)) continue;
  1012→                let resolved = candidate;
  1013→
  1014→                if (isProviderArchiveGetLinkUrl(candidate)) {
  1015→                    resolved = await resolveProviderArchiveGetLink(candidate, response?.url || archiveUrl, hints);
  1016→                }
  1017→
  1018→                if (!resolved) continue;
  1019→                if (isProviderArchiveWrapperUrl(resolved)) {
  1020→                    const nested = await resolveProviderArchiveWrapper(resolved, hints, depth + 1, visited);
  1021→                    if (nested) return nested;
  1022→                    continue;
  1023→                }
  1024→                if (isProviderArchiveGetLinkUrl(resolved)) {
  1025→                    const nested = await resolveProviderArchiveGetLink(resolved, response?.url || archiveUrl, hints);
  1026→                    if (nested) return nested;
  1027→                    continue;
  1028→                }
  1029→                return resolved;
  1030→            }
  1031→        }

  1159→function decryptMkvDramaDynamicPayload(encData, dataPath = '') {
  1160→    if (!encData?.d || !encData?.s || !dataPath) return '';
  1161→
  1162→    try {
  1163→        const normalizedPath = dataPath.startsWith('/') ? dataPath : `/${dataPath}`;
  1164→        const material = `access-payload:${normalizedPath}`;
  1165→        const key = crypto.createHash('sha256').update(material).digest();
  1166→        const iv = Buffer.from(encData.s, 'hex');
  1167→        const raw = Buffer.from(encData.d, 'base64');
  1168→        const authTag = raw.subarray(raw.length - 16);
  1169→        const ciphertext = raw.subarray(0, raw.length - 16);
  1170→        const decipher = crypto.createDecipheriv('aes-256-gcm', key, iv);
  1171→        decipher.setAuthTag(authTag);
  1172→        return Buffer.concat([decipher.update(ciphertext), decipher.final()]).toString('utf8');
  1173→    } catch {
  1174→        return '';
  1175→    }
  1176→}
