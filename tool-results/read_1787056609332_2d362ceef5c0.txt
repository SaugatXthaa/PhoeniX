     8→// Parse metadata from stream title and URL when the source doesn't provide it.
     9→// This enriches the display without modifying any source files or stream URLs.
    10→// Only fills in MISSING fields — never overwrites existing meta values.
    11→//
    12→// Parsed fields (from the user's metadata spec):
    13→//   - Quality: 2160p, 1080p, 720p, 480p
    14→//   - Source Type: BluRay Remux, BluRay, WebDL, WebRip, HDRip
    15→//   - Video Codecs: HEVC, x264, AVC, AV1
    16→//   - Audio Codecs: TrueHD, Atmos, DD+, DD, DTS, AAC, AC3
    17→//   - HDR: Dolby Vision, HDR10+, HDR
    18→//   - Bit Depth: 10-bit, 8-bit
    19→//   - Audio Language: hindi-english, english, hindi, etc.
    20→//   - Size: 64.04GB, 17.5GB, etc.
    21→//   - Release Group: FraMeSToR, ROEN-Ionicboy (after ~ or -)
    22→function enrichMeta(urlResult) {
    23→  const meta = { ...urlResult.meta };
    24→  const title = meta.title || '';
    25→  const url = urlResult.url?.href || '';
    26→  const titleLower = title.toLowerCase();
    27→  const urlLower = url.toLowerCase();
    28→
    29→  // 1. Parse height (Quality) from title if not in meta
    30→  if (!meta.height) {
    31→    if (/4k|2160p|uhd/i.test(title)) meta.height = 2160;
    32→    else if (/1080p/i.test(title)) meta.height = 1080;
    33→    else if (/720p/i.test(title)) meta.height = 720;
    34→    else if (/480p/i.test(title)) meta.height = 480;
    35→    else if (/360p/i.test(title)) meta.height = 360;
    36→    else {
    37→      const m = title.match(/(\d{3,4})p/i);
    38→      if (m) meta.height = parseInt(m[1]);
    39→    }
    40→  }
    41→
    42→  // 1b. If still no height, try parsing from URL path
    43→  // Common patterns: /1080/, /720/, /480/, /hls3/01/0720, /quality=1080
    44→  if (!meta.height) {
    45→    const urlHeightMatch = urlLower.match(/\/(1080|720|480|360|2160)\b/);
    46→    if (urlHeightMatch) meta.height = parseInt(urlHeightMatch[1]);
    47→  }
    48→
    49→  // 1c. If still no height and it's a video stream (HLS/MP4), default to 1080p
    50→  // Most anime/movie HLS streams are 1080p — this ensures all sources show
    51→  // a quality label in the enriched metadata.
    52→  if (!meta.height && (urlResult.format === Format.hls || urlResult.format === Format.mp4 ||
    53→      urlLower.includes('.m3u8') || urlLower.includes('/hls') || urlLower.includes('.mp4') || urlLower.includes('.mkv'))) {
    54→    meta.height = 1080;
    55→  }
    56→
    57→  // 2. Parse video codec from title if not in meta
    58→  if (!meta.codec && !meta.codecs) {
    59→    if (/\bhevc\b|\bx265\b|\bh\.?265\b/i.test(title)) meta.codec = 'HEVC';
    60→    else if (/\bx264\b|\bh264\b|\bavc\b/i.test(title)) meta.codec = 'AVC';
    61→    else if (/\bav1\b/i.test(title)) meta.codec = 'AV1';
    62→  }
    63→
    64→  // 3. Parse source type from title (BluRay Remux, WebDL, etc.)
    65→  if (!meta.sourceType) {
    66→    if (/bluRay\s*remux|remux/i.test(title)) meta.sourceType = 'BluRay Remux';
    67→    else if (/bluRay|bluray|bdrip/i.test(title)) meta.sourceType = 'BluRay';
    68→    else if (/web\s*dl|web-dl|webdl/i.test(title)) meta.sourceType = 'WebDL';
    69→    else if (/web\s*rip|webrip/i.test(title)) meta.sourceType = 'WebRip';
    70→    else if (/hd\s*rip|hdrip/i.test(title)) meta.sourceType = 'HDRip';
    71→    else if (/dvdrip/i.test(title)) meta.sourceType = 'DVDRip';
    72→    else if (/cam|ts\s*rip|tsrip/i.test(title)) meta.sourceType = 'CAM';
    73→    // Also catch "WEB" as a standalone word (e.g., "WEB h265", "NF WEB")
    74→    // and "NF" (Netflix), "AMZN" (Amazon), "ATVP" (Apple TV+)
    75→    else if (/\bWEB\b/i.test(title)) meta.sourceType = 'WebDL';
    76→    else if (/\bNF\b/i.test(title)) meta.sourceType = 'WebDL';
    77→    else if (/\bAMZN\b/i.test(title)) meta.sourceType = 'WebDL';
    78→    else if (/\bATVP\b/i.test(title)) meta.sourceType = 'WebDL';
    79→    else if (/\biTunes\b/i.test(title)) meta.sourceType = 'WebDL';
    80→    // Provider indicators — when the stream comes from a known streaming
    81→    // provider, the source type is WebDL (streaming-rip)
    82→    else if (/Provider:\s*CDN|Provider:\s*m4uhd|Provider:\s*lamovie|Provider:\s*1movies|Provider:\s*superflix|Provider:\s*mb-flix/i.test(title)) meta.sourceType = 'WebDL';
    83→    // HLS streams from speedracelight/VidEasy/VidKing — these are streaming rips
    84→    else if (urlLower.includes('speedracelight') || urlLower.includes('ironwallnet') || urlLower.includes('vimeos')) meta.sourceType = 'WebDL';
    85→    // Direct Google Drive / googleusercontent — typically WebDL rips
    86→    else if (urlLower.includes('googleusercontent.com') || urlLower.includes('driveseed.org')) meta.sourceType = 'WebDL';
    87→    // Cloudflare R2 / pub-*.r2.dev — typically WebDL rips (CineFreak, Movies4u)
    88→    else if (urlLower.includes('.r2.dev') || urlLower.includes('.r2.cloudflarestorage.com')) meta.sourceType = 'WebDL';
    89→    // streamraiwind.stream — HLS streaming rips (HDGharTV)
    90→    else if (urlLower.includes('streamraiwind')) meta.sourceType = 'WebDL';
    91→    // Workers.dev proxy URLs — streaming rips (ZXCStream, Pantyflix)
    92→    // Also catch proxy URLs that wrap workers.dev URLs
    93→    else if ((urlLower.includes('.workers.dev') || urlLower.includes('devcorp.me')) && !urlLower.includes('/proxy?')) meta.sourceType = 'WebDL';
    94→    else if (urlLower.includes('workers.dev') && urlLower.includes('/proxy?')) meta.sourceType = 'WebDL';
    95→    // HLS/MP4 from known streaming CDNs — these are streaming rips
    96→    else if (urlLower.includes('mycdn-mb.xyz') || urlLower.includes('scalableimpactgroup') || urlLower.includes('strategicgrowthpartners') || urlLower.includes('fsharetv') || urlLower.includes('komiknostalgia') || urlLower.includes('dropcdn') || urlLower.includes('serversicuro') || urlLower.includes('gxplayer')) meta.sourceType = 'WebDL';
    97→    // HubCloud CDN (4KHDHub pixel.hubcloud.cx) — typically WebDL
    98→    else if (urlLower.includes('hubcloud.cx')) meta.sourceType = 'WebDL';
    99→  }
   100→
   101→  // 4. Parse audio codec from title
   102→  if (!meta.audioCodec) {
   103→    if (/truehd/i.test(title)) meta.audioCodec = 'TrueHD';
   104→    else if (/atmos/i.test(title)) meta.audioCodec = 'Atmos';
   105→    else if (/dd\+|ddp|eac3/i.test(title)) meta.audioCodec = 'DD+';
   106→    else if (/\bdd\b|\bac3\b|dolby\s*digital\b/i.test(title)) meta.audioCodec = 'DD';
   107→    else if (/\bdts\b/i.test(title)) meta.audioCodec = 'DTS';
   108→    else if (/\baac\b/i.test(title)) meta.audioCodec = 'AAC';
   109→  }
   110→
   111→  // 5. Parse HDR info from title
   112→  if (!meta.hdr) {
   113→    if (/dolby\s*vision|\bdv\b/i.test(title)) meta.hdr = 'Dolby Vision';
   114→    else if (/hdr10\+/i.test(title)) meta.hdr = 'HDR10+';
   115→    else if (/\bhdr\b/i.test(title)) meta.hdr = 'HDR';
   116→  }
   117→
   118→  // 6. Parse bit depth from title
   119→  if (!meta.bitDepth) {
   120→    if (/10\s*bit|10bit|10-bit/i.test(title)) meta.bitDepth = '10-bit';
   121→    else if (/8\s*bit|8bit|8-bit/i.test(title)) meta.bitDepth = '8-bit';
   122→  }
   123→
   124→  // 7. Parse file size from title if not in meta
   125→  if (!meta.bytes) {
   126→    // Match patterns like "6.7 GB", "900 MB", "1.2GB", "[410 MB]", "31.4GB"
   127→    const sizeMatch = title.match(/(\d+(?:\.\d+)?)\s*(GB|MB)/i);
   128→    if (sizeMatch) {
   129→      const val = parseFloat(sizeMatch[1]);
   130→      meta.bytes = sizeMatch[2].toUpperCase() === 'GB'
   131→        ? val * 1024 * 1024 * 1024
   132→        : val * 1024 * 1024;
   133→    }
   134→  }
   135→
   136→  // 8. Parse release group from title (after ~ or at end in parentheses)
   137→  if (!meta.releaseGroup) {
   138→    // Pattern: "Title ~GroupName" or "Title (GroupName)" or "Title - GroupName"
   139→    const tildeMatch = title.match(/~\s*([A-Za-z0-9._-]+)/);
   140→    if (tildeMatch) {
   141→      meta.releaseGroup = tildeMatch[1];
   142→    } else {
   143→      // Try parentheses at end: "Title (FraMeSToR-LUMiX)"
   144→      const parenMatch = title.match(/\(([A-Za-z0-9._-]+)\)\s*\.?\s*$/);
   145→      if (parenMatch) meta.releaseGroup = parenMatch[1];
   146→    }
   147→  }
   148→
   149→  // 8a. Parse audio channels (5.1, 7.1, 2.0, etc.)
   150→  if (!meta.audioChannels) {
   151→    const chMatch = title.match(/\b(\d(?:\.\d)?)\s*(?:ch|channels?)\b/i)
   152→      || title.match(/\b(\d(?:\.\d)?)\b(?=\s*(?:ddp|dd|truehd|dts|atmos|eac3|ac3))/i)
   153→      || title.match(/(?:ddp|dd|truehd|dts|atmos|eac3|ac3)\s*(\d(?:\.\d)?)/i);
   154→    if (chMatch) meta.audioChannels = chMatch[1];
   155→  }
   156→
   157→  // 8b. Parse streaming platform (NF, AMZN, ATVP, Hulu, Disney+, etc.)
   158→  if (!meta.streamingPlatform) {
   159→    if (/\bNF\b|\bNetflix\b/i.test(title)) meta.streamingPlatform = 'Netflix';
   160→    else if (/\bAMZN\b|\bAmazon\b/i.test(title)) meta.streamingPlatform = 'Amazon';
   161→    else if (/\bATVP\b|\bApple\s*TV/i.test(title)) meta.streamingPlatform = 'Apple TV+';
   162→    else if (/\bHulu\b/i.test(title)) meta.streamingPlatform = 'Hulu';
   163→    else if (/\bDisney/i.test(title)) meta.streamingPlatform = 'Disney+';
   164→    else if (/\bHMAX\b|\bHBO\b/i.test(title)) meta.streamingPlatform = 'HBO Max';
   165→    else if (/\bPCOK\b|\bPeacock\b/i.test(title)) meta.streamingPlatform = 'Peacock';
   166→    else if (/\bSTAN\b/i.test(title)) meta.streamingPlatform = 'Stan';
   167→    else if (/\bBCORE\b/i.test(title)) meta.streamingPlatform = 'BlueMAX';
   168→    else if (/\bSHO\b/i.test(title)) meta.streamingPlatform = 'Showtime';
   169→    else if (/\bSTARZ\b/i.test(title)) meta.streamingPlatform = 'Starz';
   170→    else if (/\bMZKT\b|\bMax\b/i.test(title)) meta.streamingPlatform = 'Max';
   171→  }
   172→
   173→  // 8c. Parse special tags (PROPER, REPACK, UNCUT, UNCENSORED, REMASTERED, etc.)
   174→  if (!meta.specialTags) {
   175→    const tags = [];
   176→    if (/\bPROPER\b/i.test(title)) tags.push('PROPER');
   177→    if (/\bREPACK\b/i.test(title)) tags.push('REPACK');
   178→    if (/\bUNCUT\b/i.test(title)) tags.push('UNCUT');
   179→    if (/\bUNCENSORED\b/i.test(title)) tags.push('UNCENSORED');
   180→    if (/\bREMASTERED\b/i.test(title)) tags.push('REMASTERED');
   181→    if (/\bHYBRID\b/i.test(title)) tags.push('HYBRID');
   182→    if (/\bDC\b|\bDIRECTOR.?S?\s*CUT\b/i.test(title)) tags.push('DC');
   183→    if (/\bSE\b|\bSPECIAL\s*EDITION\b/i.test(title)) tags.push('SE');
   184→    if (/\bEXTENDED\b/i.test(title)) tags.push('EXTENDED');
   185→    if (/\bTHEATRICAL\b/i.test(title)) tags.push('THEATRICAL');
   186→    if (/\bIMAX\b/i.test(title)) tags.push('IMAX');
   187→    if (/\bCAM\b/i.test(title) && !/\bcam\s*rip/i.test(title)) tags.push('CAM');
   188→    if (/\bSUBBED\b/i.test(title)) tags.push('SUBBED');
   189→    if (/\bDUAL\b/i.test(title)) tags.push('DUAL');
   190→    if (/\bMULTI\b/i.test(title)) tags.push('MULTI');
   191→    if (tags.length > 0) meta.specialTags = tags.join(', ');
   192→  }
   193→
   194→  // 8d. Parse resolution label (UHD, HD, FHD, etc.)
   195→  if (!meta.resolutionLabel) {
   196→    if (/\bUHD\b/i.test(title)) meta.resolutionLabel = 'UHD';
   197→    else if (/\bFHD\b/i.test(title)) meta.resolutionLabel = 'FHD';
   198→    else if (/\bHD\b/i.test(title) && !/\bHDR\b/i.test(title)) meta.resolutionLabel = 'HD';
   199→  }
   200→
   201→  // 9. Infer format from URL if not set
   202→  if (!urlResult.format || urlResult.format === Format.unknown) {
   203→    if (urlLower.includes('.m3u8') || urlLower.includes('/m3u8/') ||
   204→        urlLower.includes('/hls/') || urlLower.includes('/playlist/')) {
   205→      urlResult.format = Format.hls;
   206→    } else if (urlLower.includes('.mp4') || urlLower.includes('.mkv')) {
   207→      urlResult.format = Format.mp4;
   208→    }
   209→  }
   210→
   211→  // 10. Parse audio languages from title if countryCodes is minimal
   212→  // Use word-boundary matching to avoid false positives (e.g., "rus" in "Icarus")
   213→  if (!meta.countryCodes || meta.countryCodes.length <= 1) {
   214→    const codes = new Set(meta.countryCodes || []);
   215→    if (/\bhindi\b|\bhin\b/i.test(titleLower)) codes.add('hi');
   216→    if (/\benglish\b|\beng\b/i.test(titleLower)) codes.add('en');
   217→    if (/\bjapanese\b|\bjpn\b/i.test(titleLower)) codes.add('ja');
   218→    if (/\bkorean\b|\bkor\b/i.test(titleLower)) codes.add('ko');
   219→    if (/\bspanish\b|\besp\b/i.test(titleLower)) codes.add('es');
   220→    if (/\bfrench\b|\bfra\b/i.test(titleLower)) codes.add('fr');
   221→    if (/\btamil\b|\btam\b/i.test(titleLower)) codes.add('ta');
   222→    if (/\btelugu\b|\btel\b/i.test(titleLower)) codes.add('te');
   223→    if (/\bchinese\b|\bmandarin\b|\bchi\b/i.test(titleLower)) codes.add('zh');
   224→    if (/\brussian\b|\brus\b/i.test(titleLower)) codes.add('ru');
   225→    if (/\bgerman\b|\bger\b/i.test(titleLower)) codes.add('de');
   226→    if (codes.size > 0) meta.countryCodes = [...codes];
   227→  }
   228→
   229→  // 11. Parse sub-source name from URL hostname
   230→  // Skip if URL is a proxy URL (localhost or addon's own host)
   231→  // Skip hash-like hostnames (e.g., da194e3e41011e58ea95b0914c6212d3.example.com)
   232→  // Skip generic CDN prefixes (e.g., cdn, www, api)
   233→  // Skip Cloudflare R2 bucket IDs (pub-XXXX.r2.dev)
   234→  // Skip googleusercontent download hashes
   235→  if (!meta.serverName && !meta.subSource) {
   236→    try {
   237→      const hostname = new URL(url).hostname;
   238→      // Skip localhost/proxy URLs — they don't indicate the actual provider
   239→      if (hostname !== 'localhost' && !hostname.includes('127.0.0.1') &&
   240→          !urlLower.includes('/proxy?')) {
   241→        const shortName = hostname.replace(/^www\./, '').split('.')[0];
   242→        // Only use shortName if it's a meaningful provider name:
   243→        // - Not a hash (hex strings longer than 12 chars)
   244→        // - Not a Cloudflare R2 bucket ID (pub-XXXX)
   245→        // - Not a googleusercontent download hash (ADGPM2...)
   246→        // - Not a generic CDN/api prefix
   247→        // - Not too short (<= 2 chars)
   248→        // - Not the same as the source label
   249→        const isHash = /^[a-f0-9]{12,}$/i.test(shortName);
   250→        const isR2Bucket = /^pub-[a-f0-9]{8,}/i.test(shortName);
   251→        const isGoogleHash = /^adgpm2/i.test(shortName);
   252→        const isGeneric = ['cdn', 'api', 'www', 'static', 'media', 'video', 'stream', 'proxy'].includes(shortName.toLowerCase());
   253→        const hasCdnInName = /cdn/i.test(shortName);
   254→        // Skip random 3-letter worker subdomains (e.g., abc., ger., cvb. from
   255→        // hindmoviez workers.dev URLs) — these are random and not meaningful
   256→        const isRandomWorker = /^[a-z]{3}$/i.test(shortName) && hostname.endsWith('.workers.dev');
   257→        if (shortName && shortName.length > 2 && !isHash && !isR2Bucket && !isGoogleHash && !isGeneric && !hasCdnInName && !isRandomWorker &&
   258→            shortName.toLowerCase() !== meta.sourceLabel?.toLowerCase()) {
   259→          meta.subSource = shortName.charAt(0).toUpperCase() + shortName.slice(1);
   260→        }
   261→      }
   262→    } catch { /* not a valid URL */ }
   263→  }
   264→
   265→  urlResult.meta = meta;
   266→  return urlResult;
   267→}
   268→
   269→export class StreamResolver {
   270→  constructor(logger, extractorRegistry) {
   271→    this.logger = logger;
   272→    this.extractorRegistry = extractorRegistry;
   273→    // Dedupe concurrent stream requests — Stremio sends 2-3 duplicate
   274→    // requests for the same content in parallel. Without dedup, each
   275→    // request runs all 85 sources simultaneously (3×85=255 concurrent
   276→    // fetches), overwhelming Render's single CPU and causing sources to
   277→    // timeout/error. With dedup, the 2nd/3rd request waits for the 1st
   278→    // to complete and reuses its result.
   279→    this.inFlight = new Map();
   280→  }
   281→
   282→  async resolve(ctx, sources, type, id) {
   283→    if (sources.length === 0) {
   284→      return { streams: [{ name: 'PhoeniX', title: '⚠️ No sources found', externalUrl: ctx.hostUrl.href }] };
   285→    }
   286→
   287→    // Dedup key: type + id + season + episode (so S1E1 and S2E1 don't collide)
   288→    const dedupKey = `${type}:${id.id || id}${id.season ? `:S${id.season}:E${id.episode || 1}` : ''}`;
   289→    const existing = this.inFlight.get(dedupKey);
   290→    if (existing) {
   291→      this.logger.info(`StreamResolver: dedup hit for ${dedupKey}, reusing in-flight request`);
   292→      return existing;
   293→    }
   294→
   295→    const promise = this._resolveInternal(ctx, sources, type, id);
   296→    this.inFlight.set(dedupKey, promise);
   297→    try {
   298→      return await promise;
   299→    } finally {
   300→      this.inFlight.delete(dedupKey);
   301→    }
   302→  }
   303→
   304→  async _resolveInternal(ctx, sources, type, id) {
   305→
   306→    const streams = [];
   307→    const urlResults = [];
   308→    let sourceErrorCount = 0;
   309→
   310→    // Per-source timing data — exposed via /debug/stream for diagnostics.
   311→    // Helps identify which sources are slow or failing under load.
   312→    const sourceTimings = [];
   313→
   314→    const SOURCE_TIMEOUT_MS = 30_000;
   315→    // Limit concurrency to prevent CPU starvation on Render's free tier.
   316→    // Without this, all 85+ sources fire simultaneously, causing CPU-intensive
   317→    // sources (Cinejoy's lumen-gate-v1 crypto, ZinkMovies, etc.) to take 30s+
   318→    // and hit the SOURCE_TIMEOUT.
   319→    //
   320→    // IMPORTANT: queue time counts against the source timeout. If a source waits
   321→    // 15s in the queue, it only has 15s left to run before timing out. With a
   322→    // limit of 15 (down from 20), sources wait less time in the queue, giving
   323→    // them more actual execution time. Cinejoy needs ~6s of actual execution,
   324→    // so a 15s limit gives it ~15s of slack for queue + execution.
   325→    const MAX_CONCURRENT_SOURCES = 15;
   326→
   327→    // Priority sources — these are started FIRST, before other sources, so they
   328→    // don't get stuck waiting in the queue behind 85+ other sources. Without this,
   329→    // Cinejoy (which needs ~6s of CPU) would wait 10-15s in the queue, leaving
   330→    // only 15-20s for execution — tight enough that it sometimes times out.
   331→    const PRIORITY_SOURCE_IDS = new Set(['cinejoy', 'zinkmovies', '4khdhub', 'playimdb']);
   332→    const sortedSources = [...sources].sort((a, b) => {
   333→      const aPriority = PRIORITY_SOURCE_IDS.has(a.id) ? 0 : 1;
   334→      const bPriority = PRIORITY_SOURCE_IDS.has(b.id) ? 0 : 1;
   335→      return aPriority - bPriority;
   336→    });
   337→
   338→    let activeCount = 0;
   339→    const waitQueue = [];
   340→
   341→    const withTimeout = (promise, ms, sourceId) => {
   342→      let timer;
   343→      const timeout = new Promise((_resolve, reject) => {
   344→        timer = setTimeout(() => reject(new Error(`source ${sourceId} timed out after ${ms}ms`)), ms);
   345→      });
   346→      return Promise.race([promise, timeout]).finally(() => clearTimeout(timer));
   347→    };
   348→
   349→    const handleSource = async (source) => {
   350→      // Concurrency gate: wait if too many sources are already running
   351→      const queueStart = Date.now();
   352→      if (activeCount >= MAX_CONCURRENT_SOURCES) {
   353→        await new Promise(resolve => waitQueue.push(resolve));
   354→      }
   355→      const queueTime = Date.now() - queueStart;
   356→      activeCount++;
   357→
   358→      const start = Date.now();
   359→      let status = 'ok';
   360→      let resultCount = 0;
   361→      try {
   362→        const sourceResults = await withTimeout(source.handle(ctx, type, id), SOURCE_TIMEOUT_MS, source.id);
   363→        resultCount = sourceResults.length;
   364→        this.logger.info(`Source ${source.id} returned ${sourceResults.length} results`);
   365→        const sourceUrlResults = await Promise.all(
   366→          sourceResults.map(({ url, meta, requestHeaders }) =>
   367→            this.extractorRegistry.handle(ctx, url, { sourceLabel: source.label, sourceId: source.id, priority: source.priority, ...(requestHeaders && { requestHeaders }), ...meta }, true)
   368→              .catch(e => {
   369→                const msg = e?.message || e?.constructor?.name || String(e);
   370→                this.logger.warn(`Extractor for ${source.id} ${url.href} error: ${msg}`);
   371→                return [];
   372→              })
   373→          )
   374→        );
   375→        urlResults.push(...sourceUrlResults.flat());
   376→      } catch (error) {
   377→        status = error?.message?.includes('timed out') ? 'timeout' : 'error';
   378→        sourceErrorCount++;
   379→        const msg = error?.message || error?.constructor?.name || String(error);
   380→        this.logger.warn(`Source ${source.id} error: ${msg}`);
   381→      } finally {
   382→        const duration = Date.now() - start;
   383→        sourceTimings.push({
   384→          id: source.id,
   385→          status,
   386→          count: resultCount,
   387→          durationMs: duration,
   388→          queueMs: queueTime,
   389→        });
   390→        activeCount--;
   391→        // Start next waiting source if any
   392→        const next = waitQueue.shift();
   393→        if (next) next();
   394→      }
   395→    };
   396→
   397→    await Promise.all(sortedSources.map(s => handleSource(s)));
   398→
   399→    // Stash timings on the instance for the /debug/stream endpoint to read.
   400→    // (Not returned in the normal /stream response to avoid breaking Stremio.)
   401→    this._lastSourceTimings = sourceTimings;
   402→
   403→    // Enrich metadata for all results (parse from title/URL — no source changes)
   404→    for (const r of urlResults) {
   405→      if (!r.error) enrichMeta(r);
   406→    }
   407→
   408→    // Sort: errors first, then by height desc, then bytes desc, then priority
   409→    urlResults.sort((a, b) => {
   410→      if (a.error || b.error) return a.error ? -1 : 1;
   411→      if (a.isExternal || b.isExternal) return a.isExternal ? 1 : -1;
   412→      const h = (b.meta?.height ?? 0) - (a.meta?.height ?? 0);
   413→      if (h !== 0) return h;
   414→      const bs = (b.meta?.bytes ?? 0) - (a.meta?.bytes ?? 0);
   415→      if (bs !== 0) return bs;
   416→      return (b.meta?.priority ?? 0) - (a.meta?.priority ?? 0);
   417→    });
   418→
   419→    // Build streams
   420→    const seen = new Set();
   421→    for (const urlResult of urlResults) {
   422→      if (urlResult.error) continue;
   423→
   424→      // Dedup by URL + sourceId — allows the same URL from different sources
   425→      // (e.g., HiAnime and AnimeKai both use zokoanime.video backend)
   426→      const urlKey = `${urlResult.url.href}__${urlResult.meta?.sourceId || ''}`;
   427→      if (seen.has(urlKey)) continue;
   428→      seen.add(urlKey);
   429→
   430→      // Route CDN URLs through /proxy if they're not already proxied
   431→      // and don't have proxyHeaders set. CDN servers (cdn.valentine.guru,
   432→      // cdn.fukggl.buzz, etc.) often reset connections when Stremio's
   433→      // ffmpeg player fetches them directly.
   434→      let finalUrl = urlResult.url;
   435→      let finalMeta = urlResult.meta;
   436→      const isAlreadyProxied = finalUrl.href.includes('/proxy?');
   437→      const hasProxyHeaders = !!urlResult.requestHeaders;
   438→      // Route URLs through /proxy ONLY if they are known to fail with direct access.
   439→      // Proxying everything causes "network connection was lost" on Render when
   440→      // downloading large files — Render kills long-running proxy connections.
   441→      // Only proxy CDNs that return "Connection reset by peer" to Stremio's player.
   442→      //
   443→      // IMPORTANT: hakunaymatata.com is NOT in this list because VidLink uses
   444→      // bcdn.hakunaymatata.com for direct MP4 streams that play fine without
   445→      // proxy. MovieBox URLs (which DO need proxy) have requestHeaders set and
   446→      // are handled by the hasProxyHeaders block below.
   447→      const needsProxy = /valentine|fukggl|fileserver|animeheaven|pixel\.hubcloud|workers\.dev/.test(finalUrl.hostname);
   448→
   449→      if (!isAlreadyProxied && !hasProxyHeaders && needsProxy) {
   450→        const proxyUrl = new URL('/proxy', ctx.hostUrl);
   451→        proxyUrl.searchParams.set('url', finalUrl.href);
   452→        // Add Referer for HubCloud CDN (pixel.hubcloud.cx → workers.dev redirect chain)
   453→        if (/pixel\.hubcloud|workers\.dev/.test(finalUrl.hostname)) {
   454→          proxyUrl.searchParams.set('referer', 'https://hubcloud.cx/');
   455→        }
   456→        finalUrl = proxyUrl;
   457→      }
   458→
   459→      // hakunaymatata.com (MovieBox CDN) returns 429 (Too Many Requests) when
   460→      // accessed without a Referer. If the source set requestHeaders with a
   461→      // Referer, route through /proxy WITH the Referer param so the proxy
   462→      // sends it. This avoids the 429 rate limit.
   463→      if (!isAlreadyProxied && hasProxyHeaders && /hakunaymatata/.test(finalUrl.hostname)) {
   464→        const proxyUrl = new URL('/proxy', ctx.hostUrl);
   465→        proxyUrl.searchParams.set('url', finalUrl.href);
   466→        const referer = urlResult.requestHeaders.Referer || urlResult.requestHeaders.referer || 'https://movie-box.co/';
   467→        proxyUrl.searchParams.set('referer', referer);
   468→        finalUrl = proxyUrl;
   469→      }
   470→
   471→      const stream = {
   472→        ...(urlResult.isExternal ? { externalUrl: finalUrl.href } : { url: finalUrl.href }),
   473→        name: this.buildName(urlResult),
   474→        title: this.buildTitle(urlResult),
   475→        behaviorHints: {
   476→          bingeGroup: `phoenix-${urlResult.meta?.sourceId}-${urlResult.meta?.extractorId}`,
   477→          ...(urlResult.format !== Format.mp4 && urlResult.notWebReady !== false && { notWebReady: true }),
   478→          ...(urlResult.requestHeaders && {
   479→            notWebReady: true,
   480→            proxyHeaders: { request: urlResult.requestHeaders },
   481→          }),
   482→          ...(urlResult.meta?.bytes && { videoSize: urlResult.meta.bytes }),
   483→        },
   484→      };
   485→      streams.push(stream);
   486→    }
   487→
   488→    this.logger.info(`Returning ${streams.length} streams`);
   489→
   490→    return { streams };
   491→  }
   492→
   493→  buildUrl(urlResult) {
   494→    if (urlResult.isExternal) return { externalUrl: urlResult.url.href };
   495→    return { url: urlResult.url.href };
   496→  }

   498→  buildName(urlResult) {
   499→    const meta = urlResult.meta || {};
   500→    const parts = ['🐦‍🔥 PhoeniX'];
   501→
   502→    // Quality label (phoenix emoji for all resolutions)
   503→    const height = meta.height;
   504→    if (height >= 2160) parts.push('4K');
   505→    else if (height >= 1080) parts.push('1080p');
   506→    else if (height >= 720) parts.push('720p');
   507→    else if (height >= 480) parts.push('480p');
   508→    else if (height > 0) parts.push(getClosestResolution(height));
   509→
   510→    // Source label + subsource (server name, provider, etc.)
   511→    if (meta.sourceLabel) {
   512→      const subSource = meta.serverName || meta.provider || meta.subSource;
   513→      if (subSource && subSource !== meta.sourceLabel) {
   514→        parts.push(`${meta.sourceLabel} · ${subSource}`);
   515→      } else {
   516→        parts.push(meta.sourceLabel);
   517→      }
   518→    }
   519→
   520→    // Download indicator for file-hosting sources
   521→    if (['4khdhub', 'moviesdrive'].includes(meta.sourceId)) {
   522→      parts.push('📥');
   523→    }
   524→
   525→    if (urlResult.isExternal) parts.push('⚠️ external');
   526→
   527→    return parts.join(' · ');
   528→  }

   530→  buildTitle(urlResult) {
   531→    const meta = urlResult.meta || {};
   532→    const titleLines = [];
   533→
   534→    // Line 1: Movie/show title (from source meta)
   535→    if (meta.title) titleLines.push(meta.title);
   536→
   537→    // Line 2: Technical specs — Quality · SourceType · Codec · AudioCodec · HDR · BitDepth
   538→    const specs = [];
   539→
   540→    // Quality
   541→    const height = meta.height;
   542→    if (height >= 2160) specs.push('2160p');
   543→    else if (height >= 1080) specs.push('1080p');
   544→    else if (height >= 720) specs.push('720p');
   545→    else if (height >= 480) specs.push('480p');
   546→    else if (height > 0) specs.push(getClosestResolution(height));
   547→
   548→    // Source Type (BluRay Remux, WebDL, etc.)
   549→    if (meta.sourceType) specs.push(meta.sourceType);
   550→
   551→    // Video Codec (HEVC, AVC, AV1)
   552→    if (meta.codec) {
   553→      specs.push(meta.codec);
   554→    } else if (meta.codecs) {
   555→      const codecStr = meta.codecs;
   556→      if (codecStr.includes('avc1')) specs.push('AVC');
   557→      else if (codecStr.includes('hvc1') || codecStr.includes('hev1')) specs.push('HEVC');
   558→      else if (codecStr.includes('av01')) specs.push('AV1');
   559→    }
   560→
   561→    // Audio Codec (TrueHD, Atmos, DD+, DTS, etc.)
   562→    if (meta.audioCodec) specs.push(meta.audioCodec);
   563→
   564→    // Audio channels (5.1, 7.1, etc.) — show with audio codec
   565→    if (meta.audioChannels) specs.push(meta.audioChannels);
   566→
   567→    // HDR (Dolby Vision, HDR10+, HDR)
   568→    if (meta.hdr) specs.push(meta.hdr);
   569→
   570→    // Bit Depth (10-bit, 8-bit)
   571→    if (meta.bitDepth) specs.push(meta.bitDepth);
   572→
   573→    // Container/stream format
   574→    if (urlResult.format === Format.hls) specs.push('HLS');
   575→    else if (urlResult.format === Format.mp4) specs.push('MP4');
   576→
   577→    // Bitrate (if available from HLS manifest)
   578→    if (meta.bandwidth) {
   579→      const Mbps = (meta.bandwidth / 1000000).toFixed(1);
   580→      specs.push(`~${Mbps} Mbps`);
   581→    }
   582→
   583→    if (specs.length > 0) titleLines.push(specs.join(' · '));
   584→
   585→    // Line 3: File size
   586→    if (meta.bytes) titleLines.push(`💾 ${bytes.format(meta.bytes)}`);
   587→
   588→    // Line 4: Audio languages
   589→    if (meta.countryCodes && meta.countryCodes.length > 0) {
   590→      const langs = meta.countryCodes
   591→        .filter(cc => cc !== 'multi')
   592→        .map(cc => languageFromCountryCode(cc))
   593→        .filter(l => l && l !== 'Multi');
   594→      if (langs.length > 0) {
   595→        titleLines.push(`Audio: ${langs.join(', ')}`);
   596→      }
   597→    }
   598→
   599→    // Line 5: Release group (if parsed from filename)
   600→    if (meta.releaseGroup) {
   601→      titleLines.push(`🏷️ ${meta.releaseGroup}`);
   602→    }
   603→
   604→    // Line 5b: Streaming platform + special tags
   605→    const extraTags = [];
   606→    if (meta.streamingPlatform) extraTags.push(meta.streamingPlatform);
   607→    if (meta.specialTags) extraTags.push(meta.specialTags);
   608→    if (meta.resolutionLabel) extraTags.push(meta.resolutionLabel);
   609→    if (extraTags.length > 0) {
   610→      titleLines.push(`📌 ${extraTags.join(' · ')}`);
   611→    }
   612→
   613→    // Line 6: Source link
   614→    const sl = meta.sourceLabel;
   615→    if (sl && sl !== urlResult.label) {
   616→      titleLines.push(`🔗 ${urlResult.label} from ${sl}`);
   617→    } else {
   618→      titleLines.push(`🔗 ${urlResult.label}`);
   619→    }
   620→
   621→    return titleLines.join('\n');
   622→  }
