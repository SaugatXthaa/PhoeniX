    17→/**
    18→ * Creates a user-visible error stream when FlareSolverr is unavailable
    19→ * @param {string} reason - Reason for unavailability ('overloaded' or 'rate_limited')
    20→ * @param {Object} options - Additional options
    21→ * @returns {Object} A Stremio-formatted error stream
    22→ */
    23→function createFlareSolverrErrorStream(reason = 'overloaded', options = {}) {
    24→    const { remaining = 0 } = options;
    25→    let title, description;
    26→
    27→    if (reason === 'rate_limited') {
    28→        title = '⏳ Rate Limit Reached';
    29→        description = `You've used your FlareSolverr quota for this hour.\n${remaining > 0 ? `${remaining} requests remaining.` : 'Please try again later.'}\n\n4KHDHub | Debrid streams unaffected`;
    30→    } else {
    31→        title = '⚠️ Server Busy';
    32→        description = 'FlareSolverr is processing many requests.\nPlease try again in a moment.\n\n4KHDHub | Debrid streams unaffected';
    33→    }
    34→
    35→    return {
    36→        name: `[HS+] Sootio\nBusy`,
    37→        title: `${title}\n${description}`,
    38→        externalUrl: 'https://github.com/sootio/stremio-addon',
    39→        behaviorHints: { notWebReady: true }
    40→    };
    41→}

    43→/**
    44→ * Validates movie year against expected year
    45→ * @param {Object} content - Content object with year
    46→ * @param {number} expectedYear - Expected year
    47→ * @returns {boolean} Whether year is valid
    48→ */
    49→function validateMovieYear(content, expectedYear) {
    50→    if (!expectedYear) {
    51→        return true; // No year to validate against
    52→    }
    53→
    54→    if (!content.year) {
    55→        return true; // No year available in content, assume valid
    56→    }
    57→
    58→    // Allow a tolerance of 1 year to account for re-releases, director's cuts etc.
    59→    if (Math.abs(content.year - expectedYear) <= 1) {
    60→        return true;
    61→    } else {
    62→        console.log(`[4KHDHub] Movie year mismatch: found ${content.year}, expected ${expectedYear} (or within 1 year)`);
    63→        return false;
    64→    }
    65→}

    67→/**
    68→ * Gets streams from 4KHDHub
    69→ * @param {string} tmdbId - TMDB ID
    70→ * @param {string} type - Content type ('movie' or 'series')
    71→ * @param {number} season - Season number (for series)
    72→ * @param {number} episode - Episode number (for series)
    73→ * @param {Object} config - Configuration object
    74→ * @param {Object} [prefetchedMeta] - Optional pre-fetched Cinemeta metadata (performance optimization)
    75→ * @returns {Promise<Array>} Array of streams
    76→ */
    77→export async function get4KHDHubStreams(tmdbId, type, season = null, episode = null, config, prefetchedMeta = null) {
    78→    try {
    79→        console.log(`[4KHDHub] Starting search for TMDB ID: ${tmdbId}, Type: ${type}${season ? `, Season: ${season}` : ''}${episode ? `, Episode: ${episode}` : ''}`);
    80→
    81→        // Extract clientIp from config for per-IP rate limiting
    82→        const clientIp = config?.clientIp || null;
    83→
    84→        // Check FlareSolverr availability early (4KHDHub depends on it for HubCloud extraction)
    85→        // Only return error stream if IP is specifically rate-limited
    86→        if (clientIp && flaresolverrManager.isIpRateLimited(clientIp)) {
    87→            const remaining = flaresolverrManager.getIpRemainingRequests(clientIp);
    88→            console.warn(`[4KHDHub] Client IP ${clientIp} rate limited (${remaining} remaining)`);
    89→            return [createFlareSolverrErrorStream('rate_limited', { remaining })];
    90→        }
    91→
    92→        let streamingLinks = [];
    93→        let pageLanguages = []; // Store languages from the page badge - must be declared here for scope
    94→
    95→        // Use pre-fetched metadata if available, otherwise fetch it (fallback for direct calls)
    96→        let cinemetaDetails = prefetchedMeta;
    97→        if (!cinemetaDetails) {
    98→            console.log(`[4KHDHub] No pre-fetched metadata, fetching from Cinemeta...`);
    99→            cinemetaDetails = await Cinemeta.getMeta(type, tmdbId);
   100→        } else {
   101→            console.log(`[4KHDHub] Using pre-fetched Cinemeta metadata: "${cinemetaDetails.name}"`);
   102→        }
   103→
   104→        if (!cinemetaDetails) {
   105→            console.log(`[4KHDHub] Could not fetch TMDB details for ID: ${tmdbId}`);
   106→            return [];
   107→        }
   108→
   109→        const year = cinemetaDetails.year ? parseInt(String(cinemetaDetails.year).split('-')[0]) : null;
   110→
   111→        console.log(`[4KHDHub] TMDB Details: ${cinemetaDetails.name} (${year || 'N/A'})`);
   112→
   113→        // PERFORMANCE FIX: Run all search strategies in parallel instead of sequentially
   114→        // This reduces search time from 5-10+ seconds to ~1-2 seconds
   115→        console.log(`[4KHDHub] Running parallel searches with multiple strategies...`);
   116→
   117→        // Build all search queries upfront
   118→        const searchQueries = [];
   119→
   120→        // Primary search using the actual title
   121→        searchQueries.push({ query: cinemetaDetails.name, strategy: 'primary' });
   122→
   123→        // Search without year
   124→        const titleWithoutYear = removeYear(cinemetaDetails.name);
   125→        if (titleWithoutYear !== cinemetaDetails.name) {
   126→            searchQueries.push({ query: titleWithoutYear, strategy: 'no-year' });
   127→        }
   128→
   129→        // Alternative title formats
   130→        const alternativeQueries = generateAlternativeQueries(
   131→            cinemetaDetails.name,
   132→            cinemetaDetails.original_title
   133→        ).filter(query => query !== cinemetaDetails.name && query !== titleWithoutYear);
   134→
   135→        alternativeQueries.forEach(query => {
   136→            searchQueries.push({ query, strategy: 'alternative' });
   137→        });
   138→
   139→        console.log(`[4KHDHub] Executing ${searchQueries.length} parallel searches...`);
   140→
   141→        // Execute all searches in parallel
   142→        const searchPromises = searchQueries.map(async ({ query, strategy }) => {
   143→            try {
   144→                const results = await scrape4KHDHubSearch(query);
   145→                console.log(`[4KHDHub] ${strategy} search "${query}" found ${results.length} results`);
   146→                return { query, strategy, results };
   147→            } catch (err) {
   148→                console.log(`[4KHDHub] ${strategy} search "${query}" failed: ${err.message}`);
   149→                return { query, strategy, results: [] };
   150→            }
   151→        });
   152→
   153→        const allSearchResults = await Promise.all(searchPromises);
   154→
   155→        // Find the best match across all search results
   156→        let bestMatch = null;
   157→        let sortedMatches = [];
   158→        let searchResults = [];
   159→
   160→        for (const { query, strategy, results } of allSearchResults) {
   161→            if (results.length > 0) {
   162→                const sorted = getSortedMatches(results, cinemetaDetails.name);
   163→                const topMatch = sorted[0];
   164→
   165→                if (topMatch && (!bestMatch || (topMatch.score || 0) > (bestMatch.score || 0))) {
   166→                    bestMatch = topMatch;
   167→                    sortedMatches = sorted;
   168→                    searchResults = results;
   169→                    const scoreDisplay = (topMatch.score !== undefined && topMatch.score !== null) ? topMatch.score.toFixed(1) : 'N/A';
   170→                    console.log(`[4KHDHub] Best match from ${strategy} search "${query}" (score: ${scoreDisplay})`);
   171→                }
   172→            }
   173→        }
   174→
   175→        if (searchResults.length === 0) {
   176→            console.log(`[4KHDHub] No search results found for any query variation`);
   177→            return [];
   178→        }
   179→
   180→        if (!bestMatch) {
   181→            console.log(`[4KHDHub] No suitable match found for: ${cinemetaDetails.name}`);
   182→            return [];
   183→        }
   184→
   185→        let downloadLinks = [];
   186→        // pageLanguages is already declared at function level
   187→
   188→        if (type === 'movie') {
   189→            // PERFORMANCE FIX: Limit year validation attempts to top 5 matches to avoid wasting time
   190→            const MAX_YEAR_VALIDATION_ATTEMPTS = 5;
   191→            let validMatch = null;
   192→            const matchesToTry = sortedMatches.slice(0, MAX_YEAR_VALIDATION_ATTEMPTS);
   193→
   194→            console.log(`[4KHDHub] Trying year validation for top ${matchesToTry.length} matches in parallel (out of ${sortedMatches.length} total)`);
   195→
   196→            // PERFORMANCE FIX: Load all candidates in parallel, pick first passing year validation
   197→            const contentPromises = matchesToTry.map(async (match) => {
   198→                const scoreDisplay = (match.score !== undefined && match.score !== null) ? match.score.toFixed(1) : 'N/A';
   199→                console.log(`[4KHDHub] Loading match: ${match.title} (score: ${scoreDisplay})`);
   200→                try {
   201→                    const content = await loadContent(match.url || match.postUrl);
   202→                    return { match, content };
   203→                } catch (err) {
   204→                    console.log(`[4KHDHub] Failed to load ${match.title}: ${err.message}`);
   205→                    return null;
   206→                }
   207→            });
   208→
   209→            const contentResults = await Promise.allSettled(contentPromises);
   210→
   211→            // Iterate in original order (by score) to pick the best valid match
   212→            for (const result of contentResults) {
   213→                if (result.status !== 'fulfilled' || !result.value) continue;
   214→                const { match, content } = result.value;
   215→
   216→                if (validateMovieYear(content, year)) {
   217→                    validMatch = match;
   218→                    downloadLinks = content.downloadLinks || [];
   219→                    pageLanguages = content.languages || [];
   220→                    console.log(`[4KHDHub] Year validation passed for ${content.title}, using this match`);
   221→                    console.log(`[4KHDHub] Page languages from badge:`, pageLanguages);
   222→                    break;
   223→                } else {
   224→                    console.log(`[4KHDHub] Movie year validation failed for ${content.title}, trying next match...`);
   225→                }
   226→            }
   227→
   228→            if (!validMatch) {
   229→                console.log(`[4KHDHub] No match passed year validation after trying ${matchesToTry.length} matches`);
   230→                return [];
   231→            }
   232→        } else if ((type === 'series' || type === 'tv') && season && episode) {
   233→            const content = await loadContent(bestMatch.url || bestMatch.postUrl);
   234→            pageLanguages = content.languages || []; // Preserve page-level languages
   235→            console.log(`[4KHDHub] Page languages from badge:`, pageLanguages);
   236→            console.log(`[4KHDHub] Looking for Season ${season}, Episode ${episode}`);
   237→            console.log(`[4KHDHub] Available episodes:`, content.episodes?.map(ep => `S${ep.season}E${ep.episode} (${ep.downloadLinks?.length || 0} links)`));
   238→
   239→            const targetEpisode = content.episodes?.find(ep =>
   240→                ep.season === parseInt(season) && ep.episode === parseInt(episode)
   241→            );
   242→
   243→            if (targetEpisode) {
   244→                console.log(`[4KHDHub] Found target episode S${targetEpisode.season}E${targetEpisode.episode} with ${targetEpisode.downloadLinks?.length || 0} links`);
   245→                downloadLinks = targetEpisode.downloadLinks || [];
   246→            } else {
   247→                console.log(`[4KHDHub] Target episode S${season}E${episode} not found`);
   248→            }
   249→        }
   250→
   251→        if (downloadLinks.length === 0) {
   252→            console.log(`[4KHDHub] No download links found`);
   253→            return [];
   254→        }
   255→
   256→        // CHECK FOR LAZY-LOAD MODE
   257→        if (isLazyLoadEnabled()) {
   258→            console.log(`[4KHDHub] Lazy-load enabled: returning ${downloadLinks.length} preview streams without extraction/validation`);
   259→
   260→            // Create preview streams from download links with metadata
   261→            const previewStreams = downloadLinks.map(link => {
   262→                // Handle both old format (string URLs) and new format (objects with metadata)
   263→                const url = typeof link === 'string' ? link : link.url;
   264→                const size = typeof link === 'object' ? link.size : null;
   265→                const quality = typeof link === 'object' ? link.quality : null;
   266→                const label = typeof link === 'object' ? link.label : 'Download';
   267→
   268→                return createPreviewStream({
   269→                    url,
   270→                    label,
   271→                    provider: '4KHDHub',
   272→                    size,
   273→                    languages: pageLanguages || []
   274→                });
   275→            });
   276→
   277→            // Sort by URL to maintain consistent order
   278→            previewStreams.sort((a, b) => {
   279→                // Prioritize certain patterns
   280→                const aPriority = a.url.toLowerCase().includes('hubcloud') || a.url.toLowerCase().includes('pixel') ? 0 :
   281→                    a.url.toLowerCase().includes('hubdrive') ? 1 :
   282→                        a.url.toLowerCase().includes('workers.dev') ? 2 : 3;
   283→                const bPriority = b.url.toLowerCase().includes('hubcloud') || b.url.toLowerCase().includes('pixel') ? 0 :
   284→                    b.url.toLowerCase().includes('hubdrive') ? 1 :
   285→                        b.url.toLowerCase().includes('workers.dev') ? 2 : 3;
   286→                return aPriority - bPriority;
   287→            });
   288→
   289→            // Format for Stremio
   290→            const streams = formatPreviewStreams(previewStreams, encodeUrlForStreaming, renderLanguageFlags);
   291→
   292→            // Limit number of preview streams
   293→            const MAX_PREVIEW_STREAMS = parseInt(process.env.MAX_4KHDHUB_PREVIEW_LINKS) || 20;
   294→            const limitedStreams = streams.slice(0, MAX_PREVIEW_STREAMS);
   295→
   296→            console.log(`[4KHDHub] Returning ${limitedStreams.length} preview streams (lazy-load mode)`);
   297→            return limitedStreams;
   298→        }
   299→
   300→        // LEGACY MODE: Full extraction and validation (slow but thorough)
   301→        console.log(`[4KHDHub] Lazy-load disabled: extracting and validating all streams (legacy mode)`);
   302→        console.log(`[4KHDHub] Found ${downloadLinks.length} redirect URLs, extracting final streams...`);
   303→
   304→        // MEMORY OPTIMIZATION: Reduce defaults and make configurable
   305→        const MAX_WORKING_LINKS = parseInt(process.env.MAX_4KHDHUB_LINKS) || 15; // Reduced from 25
   306→        const PARALLEL_BATCH_SIZE = parseInt(process.env.BATCH_SIZE) || 4; // Reduced from 8 for memory
   307→
   308→        streamingLinks = [];
   309→
   310→        // MEMORY OPTIMIZATION: Limit total links to process
   311→        const MAX_LINKS_TO_PROCESS = parseInt(process.env.MAX_4KHDHUB_PROCESS) || 30;
   312→        const linksToProcess = downloadLinks.slice(0, MAX_LINKS_TO_PROCESS);
   313→        if (downloadLinks.length > MAX_LINKS_TO_PROCESS) {
   314→            console.log(`[4KHDHub] [Memory Optimization] Limiting processing to ${MAX_LINKS_TO_PROCESS} links (out of ${downloadLinks.length})`);
   315→        }
   316→
   317→        // OPTIMIZATION: Prioritize links that are more likely to have higher quality
   318→        // Move HubCloud and direct CDN links to the front
   319→        const prioritizedLinks = [...linksToProcess].sort((a, b) => {
   320→            const aUrl = typeof a === 'string' ? a : a.url;
   321→            const bUrl = typeof b === 'string' ? b : b.url;
   322→            const aPriority = aUrl.toLowerCase().includes('hubcloud') || aUrl.toLowerCase().includes('pixel') ? 0 :
   323→                aUrl.toLowerCase().includes('hubdrive') ? 1 :
   324→                    aUrl.toLowerCase().includes('workers.dev') ? 2 : 3;
   325→            const bPriority = bUrl.toLowerCase().includes('hubcloud') || bUrl.toLowerCase().includes('pixel') ? 0 :
   326→                bUrl.toLowerCase().includes('hubdrive') ? 1 :
   327→                    bUrl.toLowerCase().includes('workers.dev') ? 2 : 3;
   328→            return aPriority - bPriority;
   329→        });
   330→
   331→        console.log(`[4KHDHub] Link prioritization: HubDrive (only), as requested`);
   332→
   333→        // Process in batches for parallel execution
   334→        for (let i = 0; i < prioritizedLinks.length && streamingLinks.length < MAX_WORKING_LINKS; i += PARALLEL_BATCH_SIZE) {
   335→            const batch = prioritizedLinks.slice(i, i + PARALLEL_BATCH_SIZE);
   336→            console.log(`[4KHDHub] Processing batch ${Math.floor(i / PARALLEL_BATCH_SIZE) + 1} (${batch.length} links)...`);
   337→
   338→            // Process batch in parallel
   339→            const batchPromises = batch.map(async (link) => {
   340→                try {
   341→                    // Extract URL from link object or use string directly
   342→                    const linkUrl = typeof link === 'string' ? link : link.url;
   343→                    const extracted = await extractStreamingLinks([linkUrl]);
   344→                    return extracted; // Array of extracted streams
   345→                } catch (err) {
   346→                    console.log(`[4KHDHub] Failed to extract link: ${err.message}`);
   347→                    return [];
   348→                }
   349→            });
   350→
   351→            const batchResults = await Promise.all(batchPromises);
   352→
   353→            // Flatten and add to streamingLinks
   354→            for (const result of batchResults) {
   355→                streamingLinks.push(...result);
   356→            }
   357→
   358→            // Clear batch results to free memory
   359→            batchResults.length = 0;
   360→            batchPromises.length = 0;
   361→
   362→            // IMPROVED: Only stop early if we have enough 4K streams OR reached max
   363→            const has4K = streamingLinks.some(s => {
   364→                const title = (s.title || s.name || '').toLowerCase();
   365→                return title.includes('2160p') || title.includes('4k');
   366→            });
   367→
   368→            if (streamingLinks.length >= MAX_WORKING_LINKS && has4K) {
   369→                console.log(`[4KHDHub] Reached ${MAX_WORKING_LINKS} working links with 4K content, stopping early`);
   370→                break;
   371→            }
   372→
   373→            // Force garbage collection hint after each batch if available
   374→            if (global.gc && i % (PARALLEL_BATCH_SIZE * 2) === 0) {
   375→                global.gc();
   376→                console.log('[4KHDHub] [Memory] GC triggered after batch processing');
   377→            }
   378→        }
   379→
   380→        // Clear prioritized links to free memory
   381→        prioritizedLinks.length = 0;
   382→        linksToProcess.length = 0;
   383→
   384→        console.log(`[4KHDHub] Extracted ${streamingLinks.length} working stream(s)`);
   385→
   386→        // Filter out suspicious AMP/redirect URLs
   387→        const filteredLinks = streamingLinks.filter(link => {
   388→            const url = link.url.toLowerCase();
   389→            const suspiciousPatterns = [
   390→                'www-google-com.cdn.ampproject.org',
   391→                'bloggingvector.shop',
   392→                'cdn.ampproject.org',
   393→            ];
   394→
   395→            const isSuspicious = suspiciousPatterns.some(pattern => url.includes(pattern));
   396→            if (isSuspicious) {
   397→                console.log(`[4KHDHub] Filtered out suspicious URL: ${link.url}`);
   398→                return false;
   399→            }
   400→            return true;
   401→        });
   402→
   403→        // Remove duplicates based on URL
   404→        const uniqueLinks = [];
   405→        const seenUrls = new Set();
   406→
   407→        for (const link of filteredLinks) {
   408→            if (!seenUrls.has(link.url)) {
   409→                seenUrls.add(link.url);
   410→                uniqueLinks.push(link);
   411→            }
   412→        }
   413→
   414→        console.log(`[4KHDHub] After URL dedup: ${uniqueLinks.length} unique links (${streamingLinks.length - filteredLinks.length} suspicious URLs filtered, ${filteredLinks.length - uniqueLinks.length} duplicates removed)`);
   415→
   416→        // Skip quality-based deduplication - keep all unique URLs
   417→        console.log(`[4KHDHub] Skipping quality dedup, keeping all ${uniqueLinks.length} unique URLs`);
   418→
   419→        // Validate URLs if DISABLE_4KHDHUB_URL_VALIDATION is false
   420→        let validatedLinks = uniqueLinks;
   421→        const disableValidation = process.env.DISABLE_4KHDHUB_URL_VALIDATION === 'true';
   422→
   423→        if (!disableValidation) {
   424→            // Check if seeking validation is specifically disabled (by default it's enabled)
   425→            const enableSeekValidation = process.env.DISABLE_4KHDHUB_SEEK_VALIDATION !== 'true';
   426→
   427→            console.log(`[4KHDHub] URL validation enabled, validating ${uniqueLinks.length} links...`);
   428→            console.log(`[4KHDHub] Seek validation ${enableSeekValidation ? 'enabled' : 'disabled'}`);
   429→
   430→            // Group links by hostname to identify trusted hosts that can skip validation
   431→            const trustedHosts = [];
   432→            const otherLinks = [];
   433→
   434→            for (const link of uniqueLinks) {
   435→                try {
   436→                    const urlObj = new URL(link.url);
   437→                    const hostname = urlObj.hostname;
   438→
   439→                    // Check if this host is in the trusted list
   440→                    const isTrustedHost = [
   441→                        'pixeldrain.dev',
   442→                        'pixeldrain.com',
   443→                        'r2.dev',
   444→                        'workers.dev',
   445→                        'hubcdn.fans',
   446→                        'googleusercontent.com'
   447→                    ].some(host => hostname.includes(host));
   448→
   449→                    if (isTrustedHost) {
   450→                        trustedHosts.push(link);
   451→                    } else {
   452→                        otherLinks.push(link);
   453→                    }
   454→                } catch {
   455→                    otherLinks.push(link); // If URL is malformed, add to other links for validation
   456→                }
   457→            }
   458→
   459→            // For trusted hosts, we can skip validation and immediately return them as valid
   460→            let validatedTrustedLinks = trustedHosts;
   461→
   462→            // Validate other links in chunks to avoid overwhelming the system
   463→            let validatedOtherLinks = [];
   464→            if (otherLinks.length > 0) {
   465→                // MEMORY OPTIMIZATION: Process validation in smaller chunks
   466→                const chunkSize = 3; // Reduced from 5 for better memory management
   467→                for (let i = 0; i < otherLinks.length; i += chunkSize) {
   468→                    const chunk = otherLinks.slice(i, i + chunkSize);
   469→                    const validationPromises = chunk.map(async (link) => {
   470→                        let result;
   471→
   472→                        if (enableSeekValidation) {
   473→                            // Check for seeking capability (range requests)
   474→                            result = await validateSeekableUrl(link.url);
   475→                        } else {
   476→                            // Use basic validation (original behavior)
   477→                            const isValid = await validateUrl(link.url);
   478→                            // Wrap basic validation in same format
   479→                            result = { isValid, filename: null };
   480→                        }
   481→
   482→                        if (!result.isValid) return null;
   483→
   484→                        // Update title with extracted filename if available, preserving language tags
   485→                        if (result.filename) {
   486→                            // Extract language information from original title
   487→                            const originalLangs = link.title.match(/\b(English|French|Spanish|German|Italian|Portuguese|Russian|Hindi|Japanese|Korean|Chinese|Arabic|Turkish|Dutch|Swedish|Norwegian|Danish|Finnish|Polish|Czech|Hungarian|Romanian|Greek|Hebrew|Thai)\b/gi) || [];
   488→
   489→                            // Create new title with filename and preserved language tags
   490→                            const langTags = originalLangs.length > 0 ? ' ' + originalLangs.join(' ') : '';
   491→                            const newTitle = result.filename + langTags;
   492→
   493→                            console.log(`[4KHDHub] Updating link title from "${link.title}" to "${newTitle}"`);
   494→                            link.title = newTitle;
   495→                        }
   496→
   497→                        return link;
   498→                    });
   499→
   500→                    const validationResults = await Promise.allSettled(validationPromises);
   501→                    const chunkValidatedLinks = validationResults
   502→                        .filter(result => result.status === 'fulfilled' && result.value !== null)
   503→                        .map(result => result.value);
   504→
   505→                    validatedOtherLinks = validatedOtherLinks.concat(chunkValidatedLinks);
   506→
   507→                    // Clear validation results to free memory
   508→                    validationResults.length = 0;
   509→                    validationPromises.length = 0;
   510→                    chunkValidatedLinks.length = 0;
   511→                }
   512→
   513→                if (enableSeekValidation) {
   514→                    console.log(`[4KHDHub] Seek validation complete: ${validatedOtherLinks.length}/${otherLinks.length} non-trusted links are seekable`);
   515→                } else {
   516→                    console.log(`[4KHDHub] Basic validation complete: ${validatedOtherLinks.length}/${otherLinks.length} non-trusted links are valid`);
   517→                }
   518→            } else {
   519→                console.log(`[4KHDHub] All links from trusted hosts, skipping validation for ${trustedHosts.length} links`);
   520→            }
   521→
   522→            validatedLinks = [...validatedTrustedLinks, ...validatedOtherLinks];
   523→            console.log(`[4KHDHub] Total validated links: ${validatedLinks.length}/${uniqueLinks.length}`);
   524→
   525→            // Clear validation arrays to free memory
   526→            trustedHosts.length = 0;
   527→            otherLinks.length = 0;
   528→            validatedTrustedLinks.length = 0;
   529→            validatedOtherLinks.length = 0;
   530→        } else {
   531→            console.log(`[4KHDHub] URL validation disabled, skipping validation`);
   532→        }
   533→
   534→        // Clear unique links to free memory
   535→        uniqueLinks.length = 0;
   536→        seenUrls.clear();
   537→
   538→        // Convert to Stremio format
   539→        const looksLikeHash = (str) => {
   540→            if (!str) return false;
   541→            const cleaned = str.replace(/[^a-fA-F0-9]/g, '');
   542→            return cleaned.length >= 16 && /^[a-f0-9]+$/i.test(cleaned);
   543→        };
   544→
   545→        const streams = validatedLinks.map(link => {
   546→            let displayTitle = link.title || link.name || '';
   547→            if (looksLikeHash(displayTitle) && link.name) {
   548→                displayTitle = link.name;
   549→            }
   550→
   551→            let resolution = getResolutionFromName(displayTitle);
   552→            // Add resolution assumption logic if no resolution is found in title
   553→            if (resolution === 'other') {
   554→                const titleLower = displayTitle.toLowerCase();
   555→                // Only assume resolution based on codec if no specific resolution was found in the title
   556→                if (titleLower.includes('h265') || titleLower.includes('hevc')) {
   557→                    resolution = '2160p'; // assume 4K for H265/HEVC only if no specific resolution found
   558→                } else if (titleLower.includes('h264')) {
   559→                    resolution = '1080p'; // assume 1080p for H264 only if no specific resolution found
   560→                } else {
   561→                    resolution = '1080p'; // default assumption
   562→                }
   563→            }
   564→            // Set resolution label properly - 2160p shows as "4k", 1080p shows as "1080p", etc.
   565→            let resolutionLabel;
   566→            if (resolution === '2160p') {
   567→                resolutionLabel = '4k';
   568→            } else if (resolution === '1080p') {
   569→                resolutionLabel = '1080p';
   570→            } else if (resolution === '720p') {
   571→                resolutionLabel = '720p';
   572→            } else if (resolution === '480p') {
   573→                resolutionLabel = '480p';
   574→            } else {
   575→                resolutionLabel = resolution; // fallback for other values
   576→            }
   577→            const size = link.size || 'N/A';
   578→            // Use page-level languages from badge if available, otherwise detect from title
   579→            let detectedLanguages;
   580→
   581→            if (pageLanguages.length > 0) {
   582→                // Use badge languages as primary source
   583→                const titleLanguages = detectLanguagesFromTitle(displayTitle);
   584→                // Merge badge and title languages, but badge takes priority
   585→                detectedLanguages = [...new Set([...pageLanguages, ...titleLanguages])];
   586→                console.log(`[4KHDHub] Languages - Badge: [${pageLanguages.join(', ')}], Title: [${titleLanguages.join(', ')}], Final: [${detectedLanguages.join(', ')}]`);
   587→            } else {
   588→                // No badge languages found, use title detection
   589→                detectedLanguages = detectLanguagesFromTitle(displayTitle);
   590→                console.log(`[4KHDHub] Using title-detected languages: [${detectedLanguages.join(', ')}]`);
   591→            }
   592→
   593→            // Convert size string to bytes for filtering
   594→            let sizeInBytes = 0;
   595→            if (link.size && typeof link.size === 'string') {
   596→                const sizeMatch = link.size.match(/([\d.]+)\s*(GB|MB|TB)/i);
   597→                if (sizeMatch) {
   598→                    const value = parseFloat(sizeMatch[1]);
   599→                    const unit = sizeMatch[2].toUpperCase();
   600→                    if (unit === 'GB') {
   601→                        sizeInBytes = value * 1024 * 1024 * 1024;
   602→                    } else if (unit === 'MB') {
   603→                        sizeInBytes = value * 1024 * 1024;
   604→                    } else if (unit === 'TB') {
   605→                        sizeInBytes = value * 1024 * 1024 * 1024 * 1024;
   606→                    }
   607→                }
   608→            }
   609→
   610→            const fileName = extractFileName(displayTitle || '');
   611→            const behaviorHints = { bingeGroup: '4khdhub-streams' };
   612→            if (fileName) {
   613→                behaviorHints.fileName = fileName;
   614→            }
   615→
   616→            return {
   617→                name: `[HS+] Sootio\n${resolutionLabel}`,
   618→                title: `${displayTitle}${renderLanguageFlags(detectedLanguages)}\n💾 ${size} | 4KHDHub`,
   619→                url: link.url ? encodeUrlForStreaming(link.url) : link.url,
   620→                _size: sizeInBytes,  // Preserve size in bytes for filtering
   621→                behaviorHints,
   622→                size: link.size,
   623→                resolution: resolution
   624→            }
   625→        });
   626→
   627→        // Sort by resolution first, then by size within each resolution group
   628→        streams.sort((a, b) => {
   629→            // Map resolution to numeric value for sorting (higher resolutions first)
   630→            const resolutionPriority = {
   631→                '2160p': 4,
   632→                '1440p': 3,
   633→                '1080p': 2,
   634→                '720p': 1,
   635→                '480p': 0,
   636→                'other': -1
   637→            };
   638→
   639→            const resolutionA = resolutionPriority[a.resolution] || 0;
   640→            const resolutionB = resolutionPriority[b.resolution] || 0;
   641→
   642→            // If resolutions are different, sort by resolution (higher first)
   643→            if (resolutionA !== resolutionB) {
   644→                return resolutionB - resolutionA;
   645→            }
   646→
   647→            // If resolutions are the same, sort by size (larger first)
   648→            // Convert sizes to MB for proper comparison
   649→            const getSizeInMB = (sizeStr) => {
   650→                if (!sizeStr) return 0;
   651→                const match = sizeStr.match(/([0-9.]+)\s*(GB|MB|TB)/i);
   652→                if (!match) return 0;
   653→                const value = parseFloat(match[1]);
   654→                const unit = match[2].toUpperCase();
   655→                if (unit === 'TB') return value * 1024 * 1024;
   656→                if (unit === 'GB') return value * 1024;
   657→                if (unit === 'MB') return value;
   658→                return 0;
   659→            };
   660→            const sizeA = getSizeInMB(a.size);
   661→            const sizeB = getSizeInMB(b.size);
   662→            return sizeB - sizeA;
   663→        });
   664→
   665→        // Additional episode filtering for series to ensure only requested episode is returned
   666→        if ((type === 'series' || type === 'tv') && season && episode) {
   667→            console.log(`[4KHDHub] Additional episode filtering: requested S${season}E${episode}`);
   668→            const requestedEpisodeRegex = new RegExp(`S0*${parseInt(season)}E0*${parseInt(episode)}|S0*${parseInt(season)}-E0*${parseInt(episode)}|\\b${parseInt(season)}x0*${parseInt(episode)}\\b|Episode[\\s-]*0*${parseInt(episode)}\\b`, 'i');
   669→
   670→            const episodeFilteredStreams = streams.filter(stream => {
   671→                // Check if episode information is in the title
   672→                const hasCorrectEpisode = requestedEpisodeRegex.test(stream.title);
   673→                if (hasCorrectEpisode) {
   674→                    console.log(`[4KHDHub] Keeping stream for S${season}E${episode}: ${stream.title}`);
   675→                    return true;
   676→                } else {
   677→                    console.log(`[4KHDHub] Filtering out stream (not S${season}E${episode}): ${stream.title}`);
   678→                    return false;
   679→                }
   680→            });
   681→
   682→            console.log(`[4KHDHub] Episode filtering: ${streams.length} -> ${episodeFilteredStreams.length} streams after filtering`);
   683→            console.log(`[4KHDHub] Returning ${episodeFilteredStreams.length} streams`);
   684→            return episodeFilteredStreams;
   685→        }
   686→
   687→        console.log(`[4KHDHub] Returning ${streams.length} streams`);
   688→
   689→        // Force garbage collection hint if available
   690→        if (global.gc) {
   691→            global.gc();
   692→            console.log('[4KHDHub] [Memory] GC triggered after stream processing complete');
   693→        }
   694→
   695→        return streams;
   696→
   697→    } catch (error) {
   698→        console.error(`[4KHDHub] Error getting streams:`, error.message);
   699→        return [];
   700→    }
   701→}
