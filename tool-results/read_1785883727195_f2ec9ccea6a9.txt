    18→/**
    19→ * Normalizes image URLs to absolute URLs
    20→ * @param {string} url - Image URL to normalize
    21→ * @returns {Promise<string>} Normalized URL
    22→ */
    23→async function normalizeImageUrl(url) {
    24→    if (!url) return '';
    25→    if (url.startsWith('//')) return 'https:' + url;
    26→    if (url.startsWith('/')) {
    27→        const domains = await getDomains();
    28→        const baseUrl = domains?.['4khdhub'] || '';
    29→        return baseUrl + url;
    30→    }
    31→    return url;
    32→}

    39→function generateIdFromUrl(url) {
    40→    try {
    41→        const urlParts = url.split('/');
    42→        const relevantPart = urlParts.find(part =>
    43→            part.length > 5 && !part.includes('4khdhub') && !part.includes('fans')

    45→        return relevantPart ? relevantPart.replace(/[^a-zA-Z0-9-]/g, '') : '';
    46→    } catch {
    47→        return '';
    48→    }

    63→/**
    64→ * Extracts just the filename from verbose metadata text
    65→ * @param {string} text - Verbose text containing filename and metadata
    66→ * @returns {string} Extracted filename or original text if not found
    67→ */
    68→export function extractFilename(text) {
    69→    if (!text) return text;
    70→
    71→    // Match filenames that can contain spaces, dots, brackets, etc.
    72→    // Starts with capital letter or digit, ends with video extension
    73→    // Includes: letters, numbers, spaces, dots, dashes, brackets, parentheses
    74→    const regex = /([A-Z0-9][\w\s.\-\[\]\(\):+,&~'!]+\.(?:mkv|mp4|avi|webm|ts|m2ts|mpg|mpeg|mov|wmv|flv|m4v))/gi;
    75→
    76→    const matches = [];
    77→    let match;
    78→    while ((match = regex.exec(text)) !== null) {
    79→        // Trim any leading/trailing whitespace from the match
    80→        const trimmed = match[1].trim();
    81→        matches.push(trimmed);
    82→    }
    83→
    84→    if (matches.length > 0) {
    85→        // Score each match to find the best filename
    86→        const scored = matches.map(filename => {
    87→            let score = filename.length; // Base score on length
    88→
    89→            // Strongly prefer filenames with quality markers
    90→            if (/\b(2160p|1080p|720p|480p|4K|UHD)\b/i.test(filename)) score += 1000;
    91→
    92→            // Prefer filenames with year
    93→            if (/\b(19|20)\d{2}\b/.test(filename)) score += 500;
    94→
    95→            // Prefer filenames with common release patterns
    96→            if (/\b(WEB-DL|WEBRip|BluRay|BRRip|HDTV|HEVC|H\.?265|H\.?264|x265|x264)\b/i.test(filename)) score += 300;
    97→
    98→            // Prefer filenames with episode markers (for series)
    99→            if (/\b(S\d{2}E\d{2}|S\d{2}\s+E\d{2}|Chapter|Episode)\b/i.test(filename)) score += 200;
   100→
   101→            // Penalize very short filenames (likely just fragments like "ESub.mkv")
   102→            if (filename.length < 20) score -= 500;
   103→
   104→            // Strongly penalize filenames that are just release group names or fragments
   105→            // Common release groups: FGT, FraMeSToR, SHORTBREHD, ESub, Joy, Telly, Strange, etc.
   106→            if (/^(ESub|FGT|SHORTBREHD|Strange|Joy|Telly|FraMeSToR|YIFY|RARBG|PSA|EVO|ION10|AMZN|NF)[\s\[\]]*\.mkv$/i.test(filename)) {
   107→                score -= 2000;
   108→            }
   109→
   110→            // Penalize filenames that are just a word + brackets + extension (e.g., "Joy [Strange].mkv")
   111→            if (/^[A-Za-z]+\s*\[[^\]]+\]\.mkv$/i.test(filename)) {
   112→                score -= 1500;
   113→            }
   114→
   115→            // Penalize single-word filenames (just group name)
   116→            if (/^[A-Za-z]+\.mkv$/i.test(filename) && filename.length < 25) {
   117→                score -= 1500;
   118→            }
   119→
   120→            return { filename, score };
   121→        });
   122→
   123→        // Sort by score (highest first) and pick the best
   124→        scored.sort((a, b) => b.score - a.score);
   125→        const bestMatch = scored[0].filename;
   126→
   127→        console.log(`[4KHDHub] Extracted filename: "${bestMatch}" (score: ${scored[0].score}) from ${matches.length} candidates`);
   128→        return bestMatch;
   129→    }
   130→
   131→    // Fallback: if no extension found, return the original text
   132→    return text;
   133→}
   134→
   135→const VIDEO_EXTENSION_REGEX = /\.(mkv|mp4|avi|webm|ts|m2ts|mpg|mpeg|mov|wmv|flv|m4v)/i;
   136→
   137→/**
   138→ * Attempts to surface the canonical dot-separated release name from a verbose label.
   139→ * Falls back to replacing whitespace with dots if no clean substring is found.
   140→ * @param {string} label - Raw label extracted from the page
   141→ * @returns {string} More readable release name (includes extension)
   142→ */
   143→export function simplifyReleaseLabel(label) {
   144→    if (!label) return label;
   145→
   146→    const dotSequenceRegex = /([A-Za-z0-9][A-Za-z0-9.\-]+?\.(?:mkv|mp4|avi|webm|ts|m2ts|mpg|mpeg|mov|wmv|flv|m4v))/gi;
   147→    const dotMatches = [];
   148→    let dotMatch;
   149→    while ((dotMatch = dotSequenceRegex.exec(label)) !== null) {
   150→        dotMatches.push(dotMatch[1]);
   151→    }
   152→
   153→    if (dotMatches.length > 0) {
   154→        dotMatches.sort((a, b) => {
   155→            const dotScoreA = (a.match(/\./g)?.length || 0);
   156→            const dotScoreB = (b.match(/\./g)?.length || 0);
   157→            if (dotScoreB !== dotScoreA) {
   158→                return dotScoreB - dotScoreA;
   159→            }
   160→            return b.length - a.length;
   161→        });
   162→        const bestDotMatch = dotMatches[0];
   163→        if ((bestDotMatch.match(/\./g)?.length || 0) >= 2 || bestDotMatch.length > 40) {
   164→            return bestDotMatch;
   165→        }
   166→    }
   167→
   168→    const match = label.match(VIDEO_EXTENSION_REGEX);
   169→    const extension = match ? match[1].toLowerCase() : null;
   170→    const baseStart = match ? label.slice(0, match.index) : label;
   171→    const base = baseStart
   172→        .trim()
   173→        .replace(/[-~]/g, ' ') // standardize separators before dot-ification
   174→        .replace(/\s+/g, ' ')
   175→        .trim();
   176→
   177→    const sanitized = base
   178→        .replace(/[^A-Za-z0-9]+/g, '.')
   179→        .replace(/\.{2,}/g, '.')
   180→        .replace(/^\.+|\.+$/g, '');
   181→
   182→    if (!sanitized) {
   183→        return extension ? `${base}.${extension}` : base;
   184→    }
   185→
   186→    return extension ? `${sanitized}.${extension}` : sanitized;
   187→}

   189→/**
   190→ * Scrapes 4KHDHub search results
   191→ * @param {string} searchQuery - Search query
   192→ * @returns {Promise<Array>} Array of search results
   193→ */
   194→export async function scrape4KHDHubSearch(searchQuery) {
   195→    // Check in-memory cache first
   196→    const cacheKey = searchQuery.toLowerCase().trim();
   197→    const cached = searchCache.get(cacheKey);
   198→    if (cached && Date.now() - cached.ts < SEARCH_CACHE_TTL) {
   199→        console.log(`[4KHDHub] Search cache hit (memory) for "${searchQuery}"`);
   200→        return cached.data;
   201→    }
   202→
   203→    try {
   204→        const domains = await getDomains();
   205→        if (!domains || !domains['4khdhub']) {
   206→            throw new Error('Failed to get domain information');
   207→        }
   208→
   209→        const baseUrl = domains['4khdhub'];
   210→
   211→        // Multiple search strategies to get more results
   212→        const searchStrategies = [
   213→            { query: searchQuery, name: 'primary' },
   214→            { query: removeYear(searchQuery), name: 'no-year' },
   215→            { query: searchQuery.split(' ').slice(0, 3).join(' '), name: 'first-three-words' },
   216→            { query: cleanTitle(searchQuery), name: 'clean-title' }
   217→        ];
   218→
   219→        // Remove duplicate queries and filter empty ones
   220→        const uniqueStrategies = [];
   221→        const seenQueries = new Set();
   222→        for (const strategy of searchStrategies) {
   223→            if (strategy.query && !seenQueries.has(strategy.query.toLowerCase())) {
   224→                seenQueries.add(strategy.query.toLowerCase());
   225→                uniqueStrategies.push(strategy);
   226→            }
   227→        }
   228→
   229→        // Try all search strategies in parallel to improve performance
   230→        const searchPromises = uniqueStrategies.map(async (strategy, index) => {
   231→            try {
   232→                const searchUrl = `${baseUrl}/?s=${encodeURIComponent(strategy.query)}`;
   233→
   234→                console.log(`Searching 4KHDHub with ${strategy.name} query: "${strategy.query}"`);
   235→                console.log(`Search URL: ${searchUrl}`);
   236→
   237→                const response = await makeRequest(searchUrl, {
   238→                    parseHTML: true,
   239→                    headers: {
   240→                        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36',
   241→                        'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,*/*;q=0.8',
   242→                        'Accept-Language': 'en-US,en;q=0.5',
   243→                        'Referer': baseUrl + '/',
   244→                    }
   245→                });
   246→
   247→                if (response.statusCode === 200) {
   248→                    const results = await parseMovieCards(response.document, baseUrl);
   249→                    console.log(`Got ${results.length} results from ${strategy.name} search`);
   250→                    return results;
   251→                } else {
   252→                    console.log(`Search with ${strategy.name} query failed with status: ${response.statusCode}`);
   253→                    return [];
   254→                }
   255→            } catch (strategyError) {
   256→                console.log(`Search strategy ${strategy.name} failed:`, strategyError.message);
   257→                return [];
   258→            }
   259→        });
   260→
   261→        // Wait for all search promises to complete
   262→        const searchResults = await Promise.all(searchPromises);
   263→        let allResults = [];
   264→        for (const results of searchResults) {
   265→            allResults = allResults.concat(results);
   266→        }
   267→
   268→        // Remove duplicate results based on title
   269→        const uniqueResults = [];
   270→        const seenTitles = new Set();
   271→        for (const result of allResults) {
   272→            const titleKey = result.title.toLowerCase();
   273→            if (!seenTitles.has(titleKey)) {
   274→                seenTitles.add(titleKey);
   275→                uniqueResults.push(result);
   276→            }
   277→        }
   278→
   279→        console.log(`Total unique results from all strategies: ${uniqueResults.length}`);
   280→
   281→        // Cache the results in memory
   282→        searchCache.set(cacheKey, { data: uniqueResults, ts: Date.now() });
   283→
   284→        return uniqueResults;
   285→    } catch (error) {
   286→        console.error('Error scraping 4KHDHub search results:', error);
   287→        throw error;
   288→    }
   289→}

   291→/**
   292→ * Helper function to parse movie cards from HTML
   293→ * @param {Object} $ - Cheerio instance
   294→ * @param {string} baseUrl - Base URL
   295→ * @returns {Promise<Array>} Array of parsed items
   296→ */
   297→async function parseMovieCards($, baseUrl) {
   298→    const items = [];
   299→
   300→    // Process movie cards from .card-grid .movie-card elements
   301→    $('.card-grid .movie-card').each((index, element) => {
   302→        const $element = $(element);
   303→
   304→        // Extract post URL from the anchor tag
   305→        const postUrl = $element.attr('href');
   306→
   307→        // Extract image from .movie-card-image img
   308→        let imageUrl = $element.find('.movie-card-image img').attr('src');
   309→        if (imageUrl) {
   310→            imageUrl = normalizeImageUrl(imageUrl);
   311→        }
   312→
   313→        // Extract alt text from img
   314→        const altText = $element.find('.movie-card-image img').attr('alt') || '';
   315→
   316→        // Extract title from .movie-card-title
   317→        const title = $element.find('.movie-card-title').text().trim();
   318→
   319→        // Extract metadata from .movie-card-meta
   320→        const metaText = $element.find('.movie-card-meta').text().trim();
   321→
   322→        // Extract year and season info from meta text
   323→        let year = undefined;
   324→        let season = undefined;
   325→
   326→        // Parse year (4-digit number)
   327→        const yearMatch = metaText.match(/(\d{4})/);
   328→        if (yearMatch) {
   329→            year = yearMatch[1];
   330→        }
   331→
   332→        // Parse season info (S01-S02, S01, etc.)
   333→        const seasonMatch = metaText.match(/S\d+(?:-S\d+)?/);
   334→        if (seasonMatch) {
   335→            season = seasonMatch[0];
   336→        }
   337→
   338→        // Extract formats from .movie-card-format spans
   339→        const formats = [];
   340→        $element.find('.movie-card-format').each((_, formatElement) => {
   341→            const format = $(formatElement).text().trim();
   342→            if (format) {
   343→                formats.push(format);
   344→            }
   345→        });
   346→
   347→        // Determine content type
   348→        const type = determineContentType(formats);
   349→
   350→        if (title && postUrl) {
   351→            // Make postUrl absolute if it's relative
   352→            const absolutePostUrl = postUrl.startsWith('http') ? postUrl : `${baseUrl}${postUrl.startsWith('/') ? '' : '/'}${postUrl}`;
   353→
   354→            // Generate ID from URL
   355→            const id = generateIdFromUrl(absolutePostUrl) || `4khdhub-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
   356→
   357→            items.push({
   358→                id,
   359→                title,
   360→                imageUrl: imageUrl || '',
   361→                postUrl: absolutePostUrl,
   362→                year,
   363→                season,
   364→                altText,
   365→                formats,
   366→                type: type.toLowerCase()
   367→            });
   368→        } else {
   369→            console.log('Skipping incomplete item:', {
   370→                hasTitle: !!title,
   371→                hasUrl: !!postUrl,
   372→                hasImage: !!imageUrl
   373→            });
   374→        }
   375→    });
   376→
   377→    console.log(`Successfully parsed ${items.length} movie cards`);
   378→    return items;
   379→}

   381→/**
   382→ * Loads content from a 4KHDHub page
   383→ * @param {string} url - Page URL
   384→ * @returns {Promise<Object>} Parsed content
   385→ */
   386→export async function loadContent(url) {
   387→    // Check in-memory cache first
   388→    const cached = contentCache.get(url);
   389→    if (cached && Date.now() - cached.ts < CONTENT_CACHE_TTL) {
   390→        console.log(`[4KHDHub] Content cache hit (memory) for ${url}`);
   391→        return cached.data;
   392→    }
   393→
   394→    return makeRequest(url, { parseHTML: true })
   395→        .then(response => {
   396→            const $ = response.document;
   397→            const rawTitle = $('h1.page-title').text().trim() || '';
   398→            const title = rawTitle.split('(')[0].trim() || '';
   399→            const poster = $('meta[property="og:image"]').attr('content') || '';
   400→            const tags = $('div.mt-2 span.badge').get().map(el => $(el).text());
   401→
   402→            // Extract year from title (e.g., "Wicked: For Good (2025)" -> 2025)
   403→            // Look for 4-digit year in parentheses at the end of the title
   404→            const yearMatch = rawTitle.match(/\((\d{4})\)\s*$/);
   405→            const year = yearMatch ? parseInt(yearMatch[1]) : null;
   406→            console.log(`[4KHDHub] Parsed year from title "${rawTitle}": ${year}`);
   407→
   408→            const description = $('div.content-section p.mt-4').text().trim() || '';
   409→            const trailer = $('#trailer-btn').attr('data-trailer-url') || '';
   410→
   411→            // Extract language information from badges
   412→            let languages = [];
   413→
   414→            // Debug: log all badges found on the page
   415→            console.log(`[4KHDHub] Scanning for language badges...`);
   416→            const allBadges = $('span.badge');
   417→            console.log(`[4KHDHub] Found ${allBadges.length} total badges on page`);
   418→
   419→            // Try multiple approaches to find language badge
   420→            // 1. Try by style attribute with teal color
   421→            const tealBadges = $('span.badge').filter((i, el) => {
   422→                const style = $(el).attr('style') || '';
   423→                return style.includes('#0d9488') || style.includes('rgb(13, 148, 136)');
   424→            });
   425→
   426→            if (tealBadges.length > 0) {
   427→                const languageText = tealBadges.first().text().trim();
   428→                console.log(`[4KHDHub] Found teal badge: "${languageText}"`);
   429→                languages = languageText.split(',').map(lang => lang.trim().toLowerCase()).filter(Boolean);
   430→            }
   431→
   432→            // 2. Fallback: search all badges for language-related text
   433→            if (languages.length === 0) {
   434→                console.log(`[4KHDHub] No teal badge found, searching all badges for language info...`);
   435→                $('span.badge').each((i, el) => {
   436→                    const text = $(el).text().trim();
   437→                    const style = $(el).attr('style') || '';
   438→                    console.log(`[4KHDHub] Badge ${i + 1}: "${text}" (style: "${style}")`);
   439→
   440→                    // Check if this badge contains language names
   441→                    if (text.match(/hindi|english|tamil|telugu|malayalam|kannada|bengali|marathi|punjabi|gujarati|urdu|spanish|french|german|italian|portuguese|chinese|japanese|korean|russian|arabic|dual|multi/i)) {
   442→                        console.log(`[4KHDHub] ✓ Badge ${i + 1} contains language info`);
   443→                        const langs = text.split(',').map(lang => lang.trim().toLowerCase()).filter(Boolean);
   444→                        languages.push(...langs);
   445→                    }
   446→                });
   447→                languages = [...new Set(languages)]; // Remove duplicates
   448→            }
   449→
   450→            console.log(`[4KHDHub] Final extracted languages from badges:`, languages);
   451→
   452→            const isMovie = tags.includes('Movies');
   453→
   454→            // Extract download links with size and quality information
   455→            let downloadLinksWithMetadata = [];
   456→
   457→            // Look for download-item divs which contain the link and metadata
   458→            const downloadItems = $('div.download-item');
   459→            console.log(`[4KHDHub] Found ${downloadItems.length} download items`);
   460→
   461→            downloadItems.each((i, item) => {
   462→                const $item = $(item);
   463→                const $link = $item.find('a').first();
   464→                const url = $link.attr('href');
   465→
   466→                if (!url || !url.trim()) {
   467→                    return;
   468→                }
   469→
   470→                // Accept all download links - extraction.js will route them correctly
   471→                // (redirect URLs like gadgetsweb.xyz don't have "hubdrive" in them yet)
   472→
   473→                // Extract metadata from the download item
   474→                const itemText = $item.text().trim();
   475→                const linkText = $link.text().trim();
   476→
   477→                // Look for size patterns (e.g., "2.5 GB", "500 MB")
   478→                let size = null;
   479→                const sizeMatch = itemText.match(/([0-9]+(?:\.[0-9]+)?)\s*(TB|GB|MB)/i);
   480→                if (sizeMatch) {
   481→                    size = `${sizeMatch[1]} ${sizeMatch[2].toUpperCase()}`;
   482→                }
   483→
   484→                // Look for quality patterns (e.g., "2160p", "1080p", "720p", "4K")
   485→                let quality = null;
   486→                const qualityMatch = itemText.match(/(\d{3,4}p|4K|2K)/i);
   487→                if (qualityMatch) {
   488→                    quality = qualityMatch[1];
   489→                }
   490→
   491→                // Extract label from the item text, removing button text like "Download HubCloud"
   492→                // Clean up the text first - remove excessive whitespace and newlines
   493→                let cleanedText = itemText.replace(/[\n\r\t]+/g, ' ').replace(/\s{2,}/g, ' ').trim();
   494→
   495→                // Remove common button text patterns
   496→                cleanedText = cleanedText.replace(/Download\s+(HubCloud|HubDrive|FSL\s*Server|FSLv?2?\s*Server|Mega|PixelServer)/gi, '').trim();
   497→
   498→                // Extract just the filename from the verbose metadata
   499→                let label = extractFilename(cleanedText);
   500→
   501→                // If no filename found with extension, try previous approach
   502→                if (label === cleanedText) {
   503→                    // Try to extract filename-like text (usually before quality/size info)
   504→                    const filenameMatch = cleanedText.match(/^([^[\]]+?)(?:\[|$)/);
   505→                    if (filenameMatch && filenameMatch[1].trim().length > 10) {
   506→                        // Found a potential filename - clean it further
   507→                        label = filenameMatch[1].trim().replace(/\s+/g, ' ');
   508→                    } else if (quality || size) {
   509→                        // If no filename found, build label from quality and size
   510→                        const parts = [];
   511→                        if (quality) parts.push(quality);
   512→                        if (size) parts.push(size);
   513→                        label = parts.join(' - ');
   514→                    } else {
   515→                        label = linkText || 'Download';
   516→                    }
   517→                }
   518→
   519→                label = simplifyReleaseLabel(label);
   520→
   521→                console.log(`[4KHDHub] Extracted: ${label} (${size || 'unknown size'})`);
   522→
   523→                downloadLinksWithMetadata.push({
   524→                    url,
   525→                    size,
   526→                    quality,
   527→                    label
   528→                });
   529→            });
   530→
   531→            // Fallback: if no download-item divs found, try old method for any download links
   532→            if (downloadLinksWithMetadata.length === 0) {
   533→                console.log('[4KHDHub] No download-item divs found, trying fallback selectors...');
   534→                const selectors = [
   535→                    'a[href*="drive"]',
   536→                    'a[href*="download"]',
   537→                    '.btn[href]',
