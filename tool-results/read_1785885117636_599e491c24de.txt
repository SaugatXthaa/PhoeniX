    81→function getCloudflareMarkers(body = '') {
    82→    const lower = (body || '').toLowerCase();
    83→    const markers = [];
    84→    if (lower.includes('cf-mitigated')) markers.push('cf-mitigated');
    85→    if (lower.includes('just a moment')) markers.push('just-a-moment');
    86→    if (lower.includes('cf_chl')) markers.push('cf_chl');
    87→    if (lower.includes('challenge-platform')) markers.push('challenge-platform');
    88→    if (lower.includes('cf-turnstile')) markers.push('cf-turnstile');
    89→    if (lower.includes('verify_turnstile')) markers.push('verify_turnstile');
    90→    if (lower.includes('security check')) markers.push('security-check');
    91→    if (lower.includes('cloudflare')) markers.push('cloudflare');
    92→    return markers;

   112→function cleanTitle(title = '') {
   113→    const parts = String(title).split(/[.\-_]/);
   114→    const qualityTags = [
   115→        'WEBRip', 'WEB-DL', 'WEB', 'BluRay', 'HDRip', 'DVDRip', 'HDTV',
   116→        'CAM', 'TS', 'R5', 'DVDScr', 'BRRip', 'BDRip', 'DVD', 'PDTV', 'HD'
   117→    ];
   118→    const audioTags = ['AAC', 'AC3', 'DTS', 'MP3', 'FLAC', 'DD5', 'EAC3', 'Atmos'];
   119→    const subTags = ['ESub', 'ESubs', 'Subs', 'MultiSub', 'NoSub', 'EnglishSub', 'HindiSub'];
   120→    const codecTags = ['x264', 'x265', 'H264', 'HEVC', 'AVC'];
   121→
   122→    const startIndex = parts.findIndex(part =>
   123→        qualityTags.some(tag => part.toLowerCase().includes(tag.toLowerCase()))
   124→    );
   125→    const endIndex = parts.findLastIndex(part =>
   126→        subTags.some(tag => part.toLowerCase().includes(tag.toLowerCase())) ||
   127→        audioTags.some(tag => part.toLowerCase().includes(tag.toLowerCase())) ||
   128→        codecTags.some(tag => part.toLowerCase().includes(tag.toLowerCase()))
   129→    );
   130→
   131→    if (startIndex !== -1 && endIndex !== -1 && endIndex >= startIndex) {
   132→        return parts.slice(startIndex, endIndex + 1).join('.');
   133→    }
   134→    if (startIndex !== -1) {
   135→        return parts.slice(startIndex).join('.');
   136→    }
   137→    return parts.slice(-3).join('.');
   138→}

   140→async function getCachedCfCookies(domain) {
   141→    if (!domain) return null;
   142→    const memCached = cfCookieCache.get(domain);
   143→    if (memCached?.cookies && memCached?.userAgent) {
   144→        if (CF_COOKIE_CACHE_TTL > 0 && memCached.timestamp) {
   145→            const age = Date.now() - memCached.timestamp;
   146→            if (age > CF_COOKIE_CACHE_TTL) {
   147→                cfCookieCache.delete(domain);
   148→            } else {
   149→                return memCached;
   150→            }
   151→        } else {
   152→            return memCached;
   153→        }
   154→    }
   155→
   156→    if (!SqliteCache.isEnabled()) {
   157→        const fileCached = await readFileCookieCache(domain);
   158→        if (fileCached?.cookies && fileCached?.userAgent) {
   159→            cfCookieCache.set(domain, fileCached);
   160→            return fileCached;
   161→        }
   162→        return null;
   163→    }
   164→
   165→    try {
   166→        const cached = await SqliteCache.getCachedRecord(CF_COOKIE_CACHE_SERVICE, `${CF_COOKIE_CACHE_PREFIX}${domain}`);
   167→        if (!cached?.data) return null;
   168→        if (CF_COOKIE_CACHE_TTL > 0) {
   169→            const updatedAt = cached.updatedAt || cached.createdAt;
   170→            if (updatedAt) {
   171→                const age = Date.now() - new Date(updatedAt).getTime();
   172→                if (age > CF_COOKIE_CACHE_TTL) return null;
   173→            }
   174→        }
   175→        if (cached.data.cookies && cached.data.userAgent) {
   176→            cfCookieCache.set(domain, cached.data);
   177→            return cached.data;
   178→        }
   179→    } catch (error) {
   180→        console.log(`[HubDrive] Failed to read CF cookie cache: ${error.message}`);
   181→    }
   182→    return null;
   183→}

   185→async function cacheCfCookies(domain, cookies, userAgent, requiresProxy = false) {
   186→    const resolvedUserAgent = userAgent || HUBCLOUD_USER_AGENT;
   187→    if (!domain || !Array.isArray(cookies) || cookies.length === 0 || !resolvedUserAgent) return;
   188→    const cookieHeader = cookies.map(c => `${c.name}=${c.value}`).join('; ');
   189→    if (!cookieHeader) return;
   190→    const data = {
   191→        cookies: cookieHeader,
   192→        userAgent: resolvedUserAgent,
   193→        timestamp: Date.now(),
   194→        requiresProxy: Boolean(requiresProxy)
   195→    };
   196→    cfCookieCache.set(domain, data);
   197→    if (!SqliteCache.isEnabled()) {
   198→        await writeFileCookieCache(domain, data);
   199→        return;
   200→    }
   201→    try {
   202→        await SqliteCache.upsertCachedMagnet({
   203→            service: CF_COOKIE_CACHE_SERVICE,
   204→            hash: `${CF_COOKIE_CACHE_PREFIX}${domain}`,
   205→            data
   206→        }, { ttlMs: CF_COOKIE_CACHE_TTL });
   207→    } catch (error) {
   208→        console.log(`[HubDrive] Failed to persist CF cookie cache: ${error.message}`);
   209→    }
   210→}

   212→async function clearCachedCfCookies(domain) {
   213→    if (!domain) return;
   214→    cfCookieCache.delete(domain);
   215→    if (!SqliteCache.isEnabled()) {
   216→        await deleteFileCookieCache(domain);
   217→        return;
   218→    }
   219→    try {
   220→        await SqliteCache.deleteCachedHash(CF_COOKIE_CACHE_SERVICE, `${CF_COOKIE_CACHE_PREFIX}${domain}`);
   221→    } catch (error) {
   222→        // Ignore cache delete errors
   223→    }
   224→}

   276→async function tryAcquireDomainLock(domain) {
   277→    if (!domain) return { acquired: false, release: null };
   278→    await ensureLockDir();
   279→    const lockPath = getDomainLockPath(domain);
   280→    try {
   281→        const handle = await fs.open(lockPath, 'wx');
   282→        await handle.close();
   283→        return {
   284→            acquired: true,
   285→            release: async () => {
   286→                try { await fs.unlink(lockPath); } catch { /* ignore */ }
   287→            }
   288→        };
   289→    } catch {
   290→        return { acquired: false, release: null };
   291→    }
   292→}

   294→async function waitForDomainLock(domain) {
   295→    if (!domain) return false;
   296→    const start = Date.now();
   297→    const lockPath = getDomainLockPath(domain);
   298→    while (Date.now() - start < FLARE_LOCK_WAIT_MS) {
   299→        try {
   300→            const stat = await fs.stat(lockPath);
   301→            if (Date.now() - stat.mtimeMs > FLARE_LOCK_TTL) {
   302→                await fs.unlink(lockPath);
   303→                return false;
   304→            }
   305→        } catch {
   306→            return false;
   307→        }
   308→        await new Promise(resolve => setTimeout(resolve, 500));
   309→    }
   310→    return true;
   311→}

   328→async function getOrCreateFlareSession(domain) {
   329→    if (!FLARESOLVERR_URL || !domain) return null;
   330→    const cached = flareSessionCache.get(domain);
   331→    if (cached && (Date.now() - cached.ts) < FLARE_SESSION_TTL) {
   332→        return cached.sessionId;
   333→    }
   334→
   335→    const sessionId = `sootio_hubdrive_${domain.replace(/\./g, '_')}`;
   336→    try {
   337→        const list = await axios.post(`${FLARESOLVERR_URL}/v1`, { cmd: 'sessions.list' }, {
   338→            timeout: 10000,
   339→            headers: { 'Content-Type': 'application/json' }
   340→        });
   341→        if (list.data?.sessions?.includes(sessionId)) {
   342→            flareSessionCache.set(domain, { sessionId, ts: Date.now() });
   343→            return sessionId;
   344→        }
   345→    } catch {
   346→        // ignore list errors
   347→    }
   348→
   349→    try {
   350→        const createBody = {
   351→            cmd: 'sessions.create',
   352→            session: sessionId
   353→        };
   354→        if (shouldUseFlareProxy(`https://${domain}`)) {
   355→            createBody.proxy = { url: FLARESOLVERR_PROXY_URL };
   356→        }
   357→        const create = await axios.post(`${FLARESOLVERR_URL}/v1`, createBody, {
   358→            timeout: 30000,
   359→            headers: { 'Content-Type': 'application/json' }
   360→        });
   361→        if (create.data?.status === 'ok') {
   362→            flareSessionCache.set(domain, { sessionId, ts: Date.now() });
   363→            return sessionId;
   364→        }
   365→    } catch (error) {
   366→        if (error.response?.data?.message?.includes('already exists')) {
   367→            flareSessionCache.set(domain, { sessionId, ts: Date.now() });
   368→            return sessionId;
   369→        }
   370→        console.log(`[HubDrive] FlareSolverr session create failed: ${error.message}`);
   371→    }
   372→
   373→    return null;
   374→}

   376→/**
   377→ * Internal function that actually calls FlareSolverr
   378→ * @param {string} url - URL to fetch
   379→ * @param {Object} headers - Request headers
   380→ * @returns {Promise<{document: Object, body: string}|null>}
   381→ */
   382→async function _doFlareSolverrRequest(url, headers = {}) {
   383→    // Check if FlareSolverr is available (not overloaded)
   384→    if (!flaresolverrManager.isAvailable()) {
   385→        const status = flaresolverrManager.getStatus();
   386→        console.warn(`[HubDrive] FlareSolverr unavailable: circuit=${status.circuitOpen}, queue=${status.queueDepth}`);
   387→        return { overloaded: true };
   388→    }
   389→
   390→    // Acquire rate limit slot
   391→    const slot = await flaresolverrManager.acquireSlot(30000);
   392→    if (!slot.acquired) {
   393→        console.warn(`[HubDrive] Could not acquire FlareSolverr slot: ${slot.reason}`);
   394→        return { overloaded: true };
   395→    }
   396→
   397→    console.log(`[HubDrive] Using FlareSolverr to bypass Cloudflare for ${url}`);
   398→    const flareTimeout = Math.max(FLARESOLVERR_TIMEOUT, 30000);
   399→    const domain = (() => {
   400→        try { return new URL(url).hostname; } catch { return null; }
   401→    })();
   402→    const sessionId = await getOrCreateFlareSession(domain);
   403→    const hasSession = Boolean(sessionId);
   404→    const lock = await tryAcquireDomainLock(domain);
   405→    if (!lock.acquired) {
   406→        const waited = await waitForDomainLock(domain);
   407→        const cached = await getCachedCfCookies(domain);
   408→        if (cached?.cookies) {
   409→            console.log(`[HubDrive] Using cookies from another process for ${domain}`);
   410→            slot.release(); // Release rate limit slot
   411→            return {
   412→                useCachedCookies: true,
   413→                cached
   414→            };
   415→        }
   416→        if (waited) {
   417→            console.log(`[HubDrive] Lock wait expired for ${domain}, continuing without lock`);
   418→        }
   419→    }
   420→
   421→    const requestOnce = async ({ useProxy = false, useSession = true } = {}) => {
   422→        const requestBody = {
   423→            cmd: 'request.get',
   424→            url,
   425→            maxTimeout: flareTimeout
   426→        };
   427→
   428→        if (useSession && sessionId) {
   429→            requestBody.session = sessionId;
   430→        }
   431→
   432→        if (!useSession && useProxy && FLARESOLVERR_PROXY_URL) {
   433→            requestBody.proxy = { url: FLARESOLVERR_PROXY_URL };
   434→        }
   435→
   436→        const response = await axios.post(`${FLARESOLVERR_URL}/v1`, requestBody, {
   437→            timeout: flareTimeout + 5000,
   438→            headers: { 'Content-Type': 'application/json' }
   439→        });
   440→
   441→        const solution = response?.data?.solution;
   442→        if (!solution?.response) {
   443→            console.log(`[HubDrive] FlareSolverr returned no response for ${url}`);
   444→            return null;
   445→        }
   446→
   447→        const body = solution.response;
   448→        const markers = getCloudflareMarkers(body);
   449→        const blockingMarkers = markers.filter(marker => marker !== 'cloudflare');
   450→        if (blockingMarkers.length > 0) {
   451→            console.log(`[HubDrive] FlareSolverr still blocked for ${url} markers=${blockingMarkers.join('|')}`);
   452→            return { blocked: true };
   453→        }
   454→
   455→        const headerUserAgent = headers['User-Agent'] || headers['user-agent'] || HUBCLOUD_USER_AGENT;
   456→        const solverUserAgent = solution.userAgent || headerUserAgent;
   457→        if (domain && solution.cookies) {
   458→            await cacheCfCookies(domain, solution.cookies, solverUserAgent, useProxy);
   459→        }
   460→
   461→        console.log(`[HubDrive] FlareSolverr success for ${url} (status: ${solution.status || 'n/a'})`);
   462→        return {
   463→            document: cheerio.load(body),
   464→            body,
   465→            url: solution.url || url,
   466→            statusCode: solution.status,
   467→            usedProxy: useProxy
   468→        };
   469→    };
   470→
   471→    try {
   472→        let result = await requestOnce({ useProxy: false, useSession: true });
   473→        if (result?.blocked) {
   474→            // Retry without session to force a new solve
   475→            result = await requestOnce({ useProxy: false, useSession: false });
   476→        }
   477→        if ((result?.blocked || result === null) && FLARESOLVERR_PROXY_URL) {
   478→            // Last resort: try via proxy with a clean session
   479→            result = await requestOnce({ useProxy: true, useSession: false });
   480→        }
   481→        if (result?.blocked || result === null) {
   482→            return null;
   483→        }
   484→        return result;
   485→    } catch (error) {
   486→        console.log(`[HubDrive] FlareSolverr error for ${url}: ${error.message}`);
   487→        // Report timeout to manager to help circuit breaker
   488→        if (error.message.includes('timeout') || error.code === 'ECONNABORTED') {
   489→            flaresolverrManager.reportTimeout();
   490→        } else {
   491→            flaresolverrManager.reportFailure();
   492→        }
   493→        if (hasSession && domain) {
   494→            flareSessionCache.delete(domain);
   495→        }
   496→        return null;
   497→    } finally {
   498→        slot.release(); // Release rate limit slot
   499→        if (lock.acquired) {
   500→            await lock.release();
   501→        }
   502→    }
   503→}

   505→/**
   506→ * Fetch a URL using FlareSolverr to bypass Cloudflare
   507→ * Includes thundering herd protection - only one FlareSolverr call per domain at a time
   508→ * @param {string} url - URL to fetch
   509→ * @param {Object} headers - Request headers
   510→ * @returns {Promise<{document: Object, body: string}|null>}
   511→ */
   512→async function fetchWithFlareSolverr(url, headers = {}) {
   513→    if (!FLARESOLVERR_URL) {
   514→        console.log('[HubDrive] FlareSolverr not configured, cannot bypass Cloudflare');
   515→        return null;
   516→    }
   517→
   518→    const domain = (() => {
   519→        try { return new URL(url).hostname; } catch { return null; }
   520→    })();
   521→
   522→    // If there's already a FlareSolverr request in progress for this domain, wait for it
   523→    const existingLock = domain ? flareSolverrLocks.get(domain) : null;
   524→    if (existingLock) {
   525→        console.log(`[HubDrive] Waiting for existing FlareSolverr request for ${domain}...`);
   526→        try {
   527→            await existingLock;
   528→            // After waiting, check if we now have cached cookies
   529→            const cached = await getCachedCfCookies(domain);
   530→            if (cached?.cookies) {
   531→                console.log(`[HubDrive] Using cookies from completed FlareSolverr request for ${domain}`);
   532→                // Return a special marker to signal caller should retry with cached cookies
   533→                return { useCachedCookies: true, cached };
   534→            }
   535→            // No cookies found after waiting - attempt our own FlareSolverr request
   536→            console.log(`[HubDrive] No cookies cached after waiting for lock on ${domain}, retrying FlareSolverr`);

   538→            // Lock failed, continue to make our own request
   539→        }
   540→    }
   541→
   542→    // Check again for lock - another request might have created one while we were checking cookies
   543→    const newLock = domain ? flareSolverrLocks.get(domain) : null;
   544→    if (newLock) {
   545→        console.log(`[HubDrive] Lock appeared while checking, waiting for ${domain}...`);
   546→        try {
   547→            await newLock;
   548→            const cached = await getCachedCfCookies(domain);
   549→            if (cached?.cookies) {
   550→                console.log(`[HubDrive] Using cookies from completed FlareSolverr request for ${domain}`);
   551→                return { useCachedCookies: true, cached };
   552→            }
   553→            console.log(`[HubDrive] No cookies cached after second wait on ${domain}, retrying FlareSolverr`);
   554→        } catch {
   555→            // continue
   556→        }
   557→    }
   558→
   559→    // Create a lock for this domain
   560→    let resolveLock;
   561→    const lockPromise = new Promise(resolve => { resolveLock = resolve; });
   562→    if (domain) {
   563→        flareSolverrLocks.set(domain, lockPromise);
   564→    }
   565→
   566→    try {
   567→        return await _doFlareSolverrRequest(url, headers);
   568→    } finally {
   569→        // Release the lock
   570→        if (domain) {
   571→            flareSolverrLocks.delete(domain);
   572→        }
   573→        resolveLock?.();
   574→    }
   575→}
